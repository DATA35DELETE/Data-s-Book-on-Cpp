Ağ Modülü

Qt'de çoğu ek paket modüller halindedir. Modül ise birden fazla kütüphaneyi barındıran paket gibidir. Qt'de bu durum genellikle kullanılıcak olan bir özelliğin modülünü "CMakeLists.txt" dosyasına katarak eklenir.
İlk olarak "agIslemleri" adında bir form projesi açalım. Ardından projenin CMake dosyasına şu ifadeleri ekliyelim;

cmake_minimum_required(VERSION 3.16)

project(agIslemleri VERSION 0.1 LANGUAGES CXX)

set(CMAKE_AUTOUIC ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

find_package(QT NAMES Qt6 Qt5 REQUIRED COMPONENTS Widgets)
find_package(Qt${QT_VERSION_MAJOR} REQUIRED COMPONENTS Widgets)

find_package(Qt6 REQUIRED COMPONENTS Network) # Network modülünü bulduk

set(PROJECT_SOURCES
        main.cpp
        mainwindow.cpp
        mainwindow.h
        mainwindow.ui
)

if(${QT_VERSION_MAJOR} GREATER_EQUAL 6)
    qt_add_executable(agIslemleri
        MANUAL_FINALIZATION
        ${PROJECT_SOURCES}
    )
# Define target properties for Android with Qt 6 as:
#    set_property(TARGET agIslemleri APPEND PROPERTY QT_ANDROID_PACKAGE_SOURCE_DIR
#                 ${CMAKE_CURRENT_SOURCE_DIR}/android)
# For more information, see https://doc.qt.io/qt-6/qt-add-executable.html#target-creation
else()
    if(ANDROID)
        add_library(agIslemleri SHARED
            ${PROJECT_SOURCES}
        )
# Define properties for Android with Qt 5 after find_package() calls as:
#    set(ANDROID_PACKAGE_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/android")
    else()
        add_executable(agIslemleri
            ${PROJECT_SOURCES}
        )
    endif()
endif()

target_link_libraries(agIslemleri PRIVATE Qt${QT_VERSION_MAJOR}::Widgets)
target_link_libraries(agIslemleri PRIVATE Qt6::Network) # Network modülünü ekledik

# Qt for iOS sets MACOSX_BUNDLE_GUI_IDENTIFIER automatically since Qt 6.1.
# If you are developing for iOS or macOS you should consider setting an
# explicit, fixed bundle identifier manually though.
if(${QT_VERSION} VERSION_LESS 6.1.0)
  set(BUNDLE_ID_OPTION MACOSX_BUNDLE_GUI_IDENTIFIER com.example.agIslemleri)
endif()
set_target_properties(agIslemleri PROPERTIES
    ${BUNDLE_ID_OPTION}
    MACOSX_BUNDLE_BUNDLE_VERSION ${PROJECT_VERSION}
    MACOSX_BUNDLE_SHORT_VERSION_STRING ${PROJECT_VERSION_MAJOR}.${PROJECT_VERSION_MINOR}
    MACOSX_BUNDLE TRUE
    WIN32_EXECUTABLE TRUE
)

include(GNUInstallDirs)
install(TARGETS agIslemleri
    BUNDLE DESTINATION .
    LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR}
    RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR}
)

if(QT_VERSION_MAJOR EQUAL 6)
    qt_finalize_executable(agIslemleri)
endif()

        
Artık ağ modülünü kullanabiliriz.

QNetworkAccessManager

QNetworkAccessManager, Qt ekosistemindeki internetten veri çekmemizi sağlayan bir sınıftır. Böylelikle dış kaynaklardan bilgisayarımıza bilgiler indirebilir hale geliriz.
Örnek amaçlı bir proje yapalım. Projemiz bu sayfadan bir önceki sayfanın kaynağını internetten indiren ve bir QTextEdit'e kayıt edip görüntüleyen bir sistem yapalım.
Tamamdır. Şimdi kodları yazabiliriz;

mainwindow.h

#ifndef MAINWINDOW_H
#define MAINWINDOW_H

#include <QMainWindow>
#include <QNetworkAccessManager>
#include <QNetworkReply>
#include <QFile>
#include <QDataStream>
#include <QTextStream>
#include <QMessageBox>
// Kullanıcağımız kütüphaneler

QT_BEGIN_NAMESPACE
namespace Ui {
class MainWindow;
}
QT_END_NAMESPACE

class MainWindow : public QMainWindow
{
    Q_OBJECT

public:
    explicit MainWindow(QWidget *parent = nullptr);
    ~MainWindow() override;

public slots:
    void sayfayiCek(); // İnternetten sayfayı çekme
    void sayfayiGoruntule(); // Sayfayı QTextEdit'e görüntüleme

private:
    Ui::MainWindow *ui;
};
#endif // MAINWINDOW_H


mainwindow.cpp

#include "mainwindow.h"
#include "./ui_mainwindow.h"

MainWindow::MainWindow(QWidget *parent)
    : QMainWindow(parent)
    , ui(new Ui::MainWindow)
{
    ui->setupUi(this);

    QFile::remove("indirilen.html"); // İlk açılışta "indirilen.html" adında dosyayı siler

    connect(ui->pushButton, &QPushButton::clicked, this, &MainWindow::sayfayiCek);
    connect(ui->pushButton_2, &QPushButton::clicked, this, &MainWindow::sayfayiGoruntule);
}

MainWindow::~MainWindow()
{
    delete ui;
}

QNetworkAccessManager *mngr; // İnternet olaylarının çoğunda bu yönetici yani QNetworkAccessManager sınıfı yer alır. Her işlemde "new" komutu ile yeniden oluşturulduğu için global bir işaretçi olarak tanımlanmıştır

void MainWindow::sayfayiCek() // Sayfayı çekiyoruz
{
    mngr = new QNetworkAccessManager; // Sınıfımızı oluşturduk
    connect(mngr, &QNetworkAccessManager::finished, this, // QNetworkAccessManager'de bir işlem başarılı bir şekilde gerçekleştiyse "finished" sinyali verilir
    [](QNetworkReply* donusDegeri) // "finished" sinyali bize QNetworkReply adında bir sınıf ile geri döner. Bu sınıf bizim işlemimizin değerini yani internetden ne çekti isek onun verisini bize QByteArray olarak verir
    {
        QFile ciktiDosyasi("indirilen.html"); // Verisini çektiğimiz sayfanın kaynağını bu sayfaya yazıyoruz

        if(ciktiDosyasi.open(QFile::OpenModeFlag::WriteOnly)) // Dosyamızı açtık
        {
            QDataStream akis(&ciktiDosyasi); // Akışımızı oluşturduk

            akis << donusDegeri->readAll(); // Değerin tamamını akışa yazdırdık
        }

        ciktiDosyasi.close(); // Dosyamızdan çıktık

        QMessageBox msg(
            QMessageBox::Icon::Information,
            "Başarılı!",
            "Başarılı bir şekilde sayfa indirildi."
        );

        msg.exec();

        // Bilgi mesajı
    });

    mngr->get(QNetworkRequest(QUrl("https://data35delete.github.io/Data-s-Book-on-Cpp/Lessons/38%20Qt%20Dosya%20İşlemleri.html")));
    // Bu şekilde ile de Get isteğimizi gönderdik
}

void MainWindow::sayfayiGoruntule() // Sayfayı görüntüliyoruz
{
    QFile indirilenDosya("indirilen.html"); // Dosya nesnemiz

    if(indirilenDosya.exists()) // Dosya indirilmiş ise
    {
        QString indirilenDosyaVerisi;

        if(indirilenDosya.open(QFile::OpenModeFlag::ReadOnly))
        {
            QTextStream akis(&indirilenDosya);

            indirilenDosyaVerisi = akis.readAll();

        }

        ui->textEdit->setHtml(indirilenDosyaVerisi);
    }
    else
    {
        QMessageBox msg(
            QMessageBox::Icon::Critical,
            "HATA!",
            "İndirilen dosya bulunamadı!"
        );

        msg.exec();

        // Bilgi mesajı
    }
}

Uygulama;

Post

Post, Get işlemi gibi dış kaynaklardan veri alıp-verişi sağlar. Farkları ise Get bunu açık yaparken Post ise bunu gizli yapar. Get ile büyük dosyalar gönderilemez çünkü karakter sınırı varken Post'da böyle bir durum söz konusu değildir.
Şimdi projemizi temizliyelim ve httpbin.org sitesini kullanarak bir Post işlemi gerçekleştirelim. Yine aynı şekilde "İsteği gönderme" ve "Görüntüle" butonları ile işlemlerimizi gerçekleştirelim.
mainwindow.h

#ifndef MAINWINDOW_H
#define MAINWINDOW_H

#include <QMainWindow>
#include <QNetworkAccessManager>
#include <QNetworkReply>
#include <QUrlQuery>
#include <QFile>
#include <QMessageBox>
// Kullanıcağımız kütüphaneler

QT_BEGIN_NAMESPACE
namespace Ui {
class MainWindow;
}
QT_END_NAMESPACE

class MainWindow : public QMainWindow
{
    Q_OBJECT

public:
    explicit MainWindow(QWidget *parent = nullptr);
    ~MainWindow() override;

public slots:
    void postIstegiAt(); // Post isteğini atma
    void goruntule(); // Post isteğini görüntüleme

private:
    Ui::MainWindow *ui;
};
#endif // MAINWINDOW_H


mainwindow.cpp

#include "mainwindow.h"
#include "./ui_mainwindow.h"

MainWindow::MainWindow(QWidget *parent)
    : QMainWindow(parent)
    , ui(new Ui::MainWindow)
{
    ui->setupUi(this);

    QFile::remove("gelenVeri.json");

    connect(ui->pushButton_2, &QPushButton::clicked, this, &MainWindow::postIstegiAt);
    connect(ui->pushButton, &QPushButton::clicked, this, &MainWindow::goruntule);
}

MainWindow::~MainWindow()
{
    delete ui;
}

QNetworkAccessManager *mngr;

void MainWindow::postIstegiAt()
{
    mngr = new QNetworkAccessManager;
    connect(mngr, &QNetworkAccessManager::finished, this,
    [](QNetworkReply *donusDegeri)
    {
        QFile gelenVeriDosya("gelenVeri.json");

        if(gelenVeriDosya.open(QFile::OpenModeFlag::WriteOnly))
        {
            QTextStream akis(&gelenVeriDosya);

            akis << donusDegeri->readAll();
        }
        gelenVeriDosya.close();

        QMessageBox msg(
            QMessageBox::Icon::Information,
            "Başarılı!",
            "Başarılı bir şekilde iste gönderildi ve veri alındı."
            );

        msg.exec();
    });

    QUrlQuery postIstekleri; // Post isteğindeki verileri ayarlıyoruz
    postIstekleri.addQueryItem("degisken1", "deger1");
    // "degisken1"; Değişkenin ismi
    // "deger1"; Değişkenin değeri
    postIstekleri.addQueryItem("degisken2", "deger2");
    postIstekleri.addQueryItem("degisken3", "deger3");

    QNetworkRequest postIstegi(QUrl("https://httpbin.org/post")); // QNetworkRequest ile Post isteklerini uyguladık
    // QUrl("https://httpbin.org/post"); Kullanılıcak olan servis
    postIstegi.setHeader(QNetworkRequest::KnownHeaders::ContentTypeHeader, "application/json"); // İstersek Header verisi yani ön bilgileri bu şekilde girebiliriz
    // QNetworkRequest::KnownHeaders::ContentTypeHeader; Ayarlanıcak olan ayar
    // "application/json"; Değeri

    // QNetworkRequest::KnownHeaders::ContentTypeHeader, bilgilerin hangi formatta gönderilceğini belirtir

    mngr->post(postIstegi, postIstekleri.toString(QUrl::FullyEncoded).toUtf8()); // Post isteğini gönderdik
}

void MainWindow::goruntule()
{
    QFile gelenVeri("gelenVeri.json");

    if(gelenVeri.exists())
    {
        QString gelenBilgiVerisi;

        if(gelenVeri.open(QFile::OpenModeFlag::ReadOnly))
        {
            QTextStream akis(&gelenVeri);

            gelenBilgiVerisi = akis.readAll();

        }

        ui->textEdit->setText(gelenBilgiVerisi);
    }
    else
    {
        QMessageBox msg(
            QMessageBox::Icon::Critical,
            "HATA!",
            "İndirilen istek bulunamadı!"
            );

        msg.exec();
    }
}

QSslconfiguration

Bu sınıf bizlere QNetworkRequest sınıfının işleminde hangi güvenlik protokolünü kullanıcağını ayarlayan bir sınıftır. İstersek TlsV1_2 yada TlsV1_3 gibi protokolleri zorlayabilir ve "Bu güvenlik sistemini kullan" diyerek özelleştirebiliriz.
mainwindow.h
            
#ifndef MAINWINDOW_H
#define MAINWINDOW_H

#include <QMainWindow>
#include <QNetworkAccessManager>
#include <QNetworkReply>
#include <QUrlQuery>
#include <QFile>
#include <QMessageBox>
#include <QSslConfiguration> // Kütüphane
// Kullanıcağımız kütüphaneler

QT_BEGIN_NAMESPACE
namespace Ui {
class MainWindow;
}
QT_END_NAMESPACE

class MainWindow : public QMainWindow
{
    Q_OBJECT

public:
    explicit MainWindow(QWidget *parent = nullptr);
    ~MainWindow() override;

public slots:
    void postIstegiAt(); // Post isteğini atma
    void goruntule(); // Post isteğini görüntüleme

private:
    Ui::MainWindow *ui;
};
#endif // MAINWINDOW_H


mainwindow.cpp

#include "mainwindow.h"
#include "./ui_mainwindow.h"

QNetworkAccessManager *mngr;

void MainWindow::postIstegiAt()
{
    mngr = new QNetworkAccessManager;

    connect(mngr, &QNetworkAccessManager::finished, this,
    [](QNetworkReply *donusDegeri)
    {
        QFile gelenVeriDosya("gelenVeri.json");

        if(gelenVeriDosya.open(QFile::OpenModeFlag::WriteOnly))
        {
            QTextStream akis(&gelenVeriDosya);

            akis << donusDegeri->readAll();
        }
        gelenVeriDosya.close();

        QMessageBox msg(
            QMessageBox::Icon::Information,
            "Başarılı!",
            "Başarılı bir şekilde iste gönderildi ve veri alındı."
            );

        msg.exec();
    });

    QUrlQuery postIstekleri;
    postIstekleri.addQueryItem("degisken1", "deger1");
    postIstekleri.addQueryItem("degisken2", "deger2");
    postIstekleri.addQueryItem("degisken3", "deger3");

    QNetworkRequest postIstegi(QUrl("https://httpbin.org/post"));
    postIstegi.setHeader(QNetworkRequest::KnownHeaders::ContentTypeHeader, "application/json");

    QSslConfiguration ssl = QSslConfiguration::defaultConfiguration(); // QSslConfiguration sınıfını oluşturduk
    ssl.setProtocol(QSsl::TlsV1_2); // Protokulu belirledik
    // TslV1_3 için QSsl::Tlsv1_3

    postIstegi.setSslConfiguration(ssl); // İsteğimizin güvenlik ayarını yaptık

    mngr->post(postIstegi, postIstekleri.toString(QUrl::FullyEncoded).toUtf8());
}
