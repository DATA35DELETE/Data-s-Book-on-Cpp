Veri Yönetmimi

QString

QString, Standart C++'daki String'in Qt karşılığıdır. Ama dolu özellik ile gelir. Örnek olarak C ve C++ programlama dillerindeki UTF-8 sıkıntısı ve kolay dönüşümleri sayesinde oldukça kullanışlıdır.
Bu sayfada veri yapılarını işliceğimiz için proje adı "veriYonetimi", tür olarak ise "Konsol" projesi açalım. Konsol bölümü için Non-Qt Project bölümünden "Plain C++ Application" seçeneğini seçelim. Ardından projemizi oluşturalım.
Başlamadan önce CMakeLists.txt dosyasına Qt'nin Core yani çekirdek paketinini kullanıcağımızı bildirmemiz gerekir;

cmake_minimum_required(VERSION 3.16)

project(veriYonetimi LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

add_executable(veriYonetimi main.cpp)

find_package(Qt6 REQUIRED COMPONENTS Core)
target_link_libraries(veriYonetimi PRIVATE Qt6::Core)

# Bildirdik

include(GNUInstallDirs)
install(TARGETS veriYonetimi
    LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR}
    RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR}
)

Ardından kodları inceleyebiliriz;

main.cpp
#include <QString> // QString Kütüphanemiz

#include <QDebug> // Qt'nin çıktı kütüphanesi

int main()
{
    QString str1 = "Selam"; // Bu şekilde oluşturulur
    qDebug() << "str1 = " << str1 ; // Bu şekilde yazılır

    str1.append(" Herkese!"); // Sonuna String ifadesi konulması
    qDebug() << "str1 = " << str1;

    QString str2 = QString("%1, %3, %2 ve %4")        // Parametreli olarak tanımalamak istersek
    .arg("Assembly", "C", "C++", "Rust");             //  Bu şekilde sırasıyla argümanları yazılır ve eklenir. Paramterdeki sıraları önemlidir
    qDebug() << "str2 = " << str2;

    str2.assign("Assembly"); // İçeriğini tamamiyle değiştirmek istersek
    qDebug() << "str2 = " << str2;

    QChar str2BirinciKarater = str2.at(1); // Aynı listeler gibi QString'lerinde elemanlarını alabiliriz. Döndürdüğü değer QChar'dır. Char türünün Qt versiyonudur
    qDebug() << "str2[1] = " << str2BirinciKarater;

    qDebug() << "str2'nin kapasitesi =" << str2.capacity(); // Kapasitesini verir

    QString str3 = "Selam!lar";
    str3.chop(3); // Sondan yazılan sayı kadar karakter siler
    qDebug() << "str3 = " << str3;

    QString str3KesilmisKismi = str3.chopped(1); // Kesimden sonraki kısmı QString olarak geri verir
    qDebug() << "str3KesilmisKismi =" << str3KesilmisKismi;

    str3.clear(); // İçeriği temizler

    qDebug() << "str3 boş mu" << str3.isEmpty();

    QString str4 = "Assembly ve C";
    qDebug() << "\"Assembly ve C\" == str4 =" << str4.contains("Assembly ve C"); // Aynı diye mi kontrol eder

    qDebug() << "str4 karakter sayısı" << str4.length(); // Karakter sayısını verir

    str4.erase(str4.begin() + 8, str4.end()); // Belirli kısımları keser
    // str4.begin() + 8; Başlangıç iteratöründen 8 eleman ileriden başla
    // str4.end(); Sona kadar
    qDebug() << "str4 =" << str4;

    qDebug() << "str4'deki 'm'nin Index numarası" << str4.indexOf('m'); // Belirtilen karakterin Index numarasını verir

    qDebug() << "str4'deki karakterlerin hepsi küçük mü" << str4.isLower(); // Karakterilerin hepsi küçük mü diye kontrol eder

    QString str4Lower = str4.toLower(); // Tüm karakterleri küçük karakter yapar ve bir QString değeri döndürür

    qDebug() << "str4Lower'deki karakterlerin hepsi küçük mü" << str4Lower.isLower();

    QString str4Upper = str4.toUpper(); // Tüm karakterleri büyük yapar

    qDebug() << "str4Upper'deki karakterlerin hepsi büyük mü" << str4Upper.isUpper(); // Karakterilerin hepsi büyük mü diye kontrol eder

    QString str5;

    qDebug() << "str5 Null mu" << str5.isNull();

    qDebug() << "str5'in maksimum karakter sayısı" << str5.max_size(); // Maksimum karakter sayısını verir

    str5.prepend("C++"); // Başına ifade ekler
    qDebug() << "str5 =" << str5;

    str5.remove("++"); // İfade çıkartır
    qDebug() << "str5 =" << str5;

    QString str6 = "QString";
    str6.replace("QS", "std::s"); // İfadeki kısımları değiştirir
    // "QS"; Orjinal değer
    // "std::s"; Yerine yazılcak değer
    qDebug() << "str6 =" << str6;

    // QString.toX, istenilen türe dönüştürür
    qDebug() << "QString(\"1234\").toInt() =" << QString("1234").toInt();

    QString str7 = "\n\n\nSelam\n";
    QString str7Arinmis = str7.trimmed(); // Baştaki ve sondaki "\n" karakterlerinden arındırır
    qDebug() << str7Arinmis;

    std::wstring stdWStr1 = L"Türkiye"; // std::wstring, std::string'ın karakter ailesinin genişletilmiş versiyonudur. Genelde İngilizce dışı karakter için kullanılır

    // QString::fromX(ifade), X türünden QString'e 'ifade'yi uygun şekilde geçirir
    qDebug() << QString::fromStdWString(stdWStr1);

    qDebug() << "QString::number(1234) =" << QString::number(1234); // Sayıları QString'e çevirir

    return 0;
}

QVariant

QVariant, duruma bağlı olarak değişebilen bir sınıftır. STL konusundaki Variant'dan farklı olarak QVariant, önceden tanımlanmış değişken türlerinin aksine o an kendisine atanmış değişken türünün değerini taşır. Genel olarak verilerinin türleri değişiklik gösterdiğinde kullanılır.
#include <QVariant> // QVariant Kütüphanemiz

#include <QDebug>

int main()
{
    QVariant cumhuriyetinIlani(1923); // QVariant sınıfından bir nesne tanımladık

    // QVariant.toX(); ile içerisindeki X türündeki veriyi çekebiliriz

    qDebug() << "Cumhuriyetin ilan edildiği yıl" << cumhuriyetinIlani.toInt();

    cumhuriyetinIlani = QVariant("29 Ekim"); // Şimdi QString bir ifade tanımladık

    qDebug() << "Cumhuriyetin ilan edildiği ay ve günü" << cumhuriyetinIlani.toString();
    // cumhuriyetinIlani.toInt(); Başka bir tanımlama yapıldığında QVariant bir önceki değeri siler ve sıfır döndürür. Şuan bir QString değer olduğu için Int değeri 0'dır

    return 0;
}

QHash ve QMap

QHash ve QMap, STL'deki Map'ten pek fazla farkları yoktur.
#include <QMap> // QMap Kütüphanemiz
#include <QHash> // QHash Kütüphanemiz

#include <QDebug>

int main()
{
    QMap<QString, int> ornekQMap;
    ornekQMap.insert("key1", 1002);
    ornekQMap.insert("key2", 9981);

    qDebug() << "key1 =" << ornekQMap["key1"];

    QHash<QString, int> ornekHash;
    ornekHash.insert("key1", 1002);
    ornekHash.insert("key2", 9981);

    qDebug() << "key1 =" << ornekHash["key1"];

    return 0;
}

QSet

QSet, QHash gibi eklenen değerden birden fazla varsa bir tane alır. QSet ise bu durumun tek türlü versiyonudur.
#include <QSet> // QSet Kütüphanemiz

#include <QDebug>

int main()
{
    QSet<int> sayilar;

    sayilar.insert(1);
    sayilar.insert(1);
    sayilar.insert(2);
    sayilar.insert(3);
    sayilar.insert(4);
    sayilar.insert(4);

    for (auto i : sayilar) {
        qDebug() << i;
    }

    return 0;
}

QByteArray

QByteArray, aynı bir liste gibi değer tutar. Farklı olarak ise elemanları Binary türündendir. Yani içerisinde ham veri tutulabilir.
#include <QByteArray> // QByteArray Kütüphanemiz

#include <QDebug>

int main()
{
    QByteArray selamYazisininHamVerisi; // Nesnemiz

    selamYazisininHamVerisi.append(0x53); // S
    selamYazisininHamVerisi.append(0x65); // e
    selamYazisininHamVerisi.append(0x6C); // l
    selamYazisininHamVerisi.append(0x61); // a
    selamYazisininHamVerisi.append(0x6D); // m

    qDebug() << selamYazisininHamVerisi.toStdString();

    return 0;
}

Qt'de Json verileri ile işlem yaparken herhangi bir kütüphane kullanılmaz. Genel mantık ise QJsonObject ile Key-Value ve QJsonDocument ile ham Json verisine dönüştürme sağlanır.

    dataDeleteBilgiler.insert("Soyismi", "Delete");

QSettings

QSettings, Qt'nin bizlere sunduğu yerel bir ayar tutma ve saklama kütüphanesidir. Ayrıca istenildiği takdirde ".ini" formatında da kayıt edilebilir.
#include <QSettings> // Json Kütüphaneleri

#include <QDebug>

int main()
{
    QSettings ayarlar("ayalar.ini", QSettings::Format::IniFormat); // Bu şekilde nesne oluşturulur

    ayarlar.setValue("user/arkaplanRenk", "mavi"); // Bu şekilde ayar tanımlanır
    ayarlar.setValue("root/arkaplanRenk", "siyah");

/*
ayarlar.ini
[user]
arkaplanRenk=mavi

[root]
arkaplanRenk=siyah
*/

    return 0;
}
