Diyaloglar

Diyalog Nedir?

Diyalog, işlem için özel olarak kullanılan ve işlem bittiğinde kaybolan ve ana sistemi bilgi veya mesaj veren bir pencere yapısıdır.
Peki neden kullanılır? Çünkü kolaylık! Grafiksel Kullanıcı Arayüzülü(GUI) uygulamalarda her bir iş için genellikle ayrı ayrı pencereler oluşturulur. Bu durum genellikle zaman kaybıdır. Özellikle kapsamlı seçim ekranları ve dosya ekranlarında bu durum oldukça yaygındır. Tabiki bu durumun çözümü diyaloglardır. Diyaloglar işletim sistemi tarafından sağlanan ve basit kodlarla tetiklenen, görüntülenen ve bilgi alıp-veren mini pencerelerdir.
Bu sayfada diyalogları görüceğimiz için "diyaloglar" adında bir proje oluşturalım.

QMessageBox

QMessageBox, bilgi verme amaçlı kullanılan diyalogtur.
mainwindow.h

#ifndef MAINWINDOW_H
#define MAINWINDOW_H

#include <QMainWindow>

QT_BEGIN_NAMESPACE
namespace Ui {
class MainWindow;
}
QT_END_NAMESPACE

class MainWindow : public QMainWindow
{
    Q_OBJECT

public:
    explicit MainWindow(QWidget *parent = nullptr);
    ~MainWindow() override;
    
public slots:
    void messageBoxGoster();
    
private:
    Ui::MainWindow *ui;
};
#endif // MAINWINDOW_H


mainwindow.cpp

#include "mainwindow.h"
#include "./ui_mainwindow.h"

#include <QMessageBox> // QMessageBox için kütüphane

MainWindow::MainWindow(QWidget *parent)
    : QMainWindow(parent)
    , ui(new Ui::MainWindow)
{
    ui->setupUi(this);

    connect(ui->pushButton, &QPushButton::clicked, this, &MainWindow::messageBoxGoster);
}

MainWindow::~MainWindow()
{
    delete ui;
}

void MainWindow::messageBoxGoster()
{
    QMessageBox mesajKutusu_KritikMesaj(
        QMessageBox::Icon::Critical, // QMessageBox'ın modlarına göre bazı simgeler sunar. İçeriğe bağlı olarak kritik, bilgilendirme, ikonsuz, soru ve uyarı modları bulunmaktadır. Critical, kritik bilgileri bildirmede kullanılır
        "Bir kritik mesajımız var!", // Mesaj kutusunun başlığının yazısı
        "Bu mesaj bir QMessageBox'ın içerisindeki kritik mesajdır..." // Mesaj kutusunun içerisinde yazıcak olan yazı
    ); // QMessageBox nesnini bu şekilde oluşturulur

    mesajKutusu_KritikMesaj.exec(); // Mesaj kutusunu göstermek için

    QMessageBox mesajKutusu_BilgilendiriciMesaj(
        QMessageBox::Icon::Information, // Information, bilgilendirme durumlarında kullanılır
        "Bir bilgilendirici mesajımız var!",
        "Bu mesaj bir QMessageBox'ın içerisindeki bilgilendirici mesajdır..."
        );

    mesajKutusu_BilgilendiriciMesaj.exec();

    QMessageBox mesajKutusu_Mesaj(
        QMessageBox::Icon::NoIcon, // NoIcon, sadece mesaj vermek için kullanılır
        "Bir mesajımız var!",
        "Bu mesaj bir QMessageBox'ın içerisindeki mesajdır..."
        );

    mesajKutusu_Mesaj.exec();

    QMessageBox mesajKutusu_SoruMesaji(
        QMessageBox::Icon::Question, // Question, genelde bir bilginin bulunanaması veya istenilen bilginin alınamamasında kaynaklı sorunları bildirmek için kullanılır
        "Bir sorunlu mesajımız var!",
        "Bu mesaj bir QMessageBox'ın içerisindeki sorunlu mesajdır..."
        );

    mesajKutusu_SoruMesaji.exec();

    QMessageBox mesajKutusu_UyarıcıMesaj(
        QMessageBox::Icon::Warning, // Warning, uyarı mesajları için kullanılır
        "Bir sorunlu mesajımız var!",
        "Bu mesaj bir QMessageBox'ın içerisindeki sorunlu mesajdır..."
        );

    mesajKutusu_UyarıcıMesaj.exec();
}

void MainWindow::fileDialogGoster()
{
    QFileDialog dosyaEkrani; // Dosya seçim ekranı oluşturma
    dosyaEkrani.exec(); // Dosya seçim ekranını gösterme

    if(!dosyaEkrani.selectedFiles().isEmpty()) // Eğer kullanıcı dosya seçim ekranında bir dosya seçilirse
    {
        qDebug() << "Seçilen dosyanın yolu; " << dosyaEkrani.selectedFiles().at(0).toStdString() << Qt::endl
        << "Seçilen dosyanın Url'si; " << dosyaEkrani.selectedUrls().at(0) << Qt::endl;
    }
    else // Seçilmesze
    {
        qDebug() << "Herhangi bir dosya seçilmedi!" << Qt::endl;
    }

    QFileDialog dosyaEkrani2; // Dosya seçim ekranı oluşturma
    dosyaEkrani2.setNameFilters({"Fotoğraflar (*.png *.jpeg *.jpg)", "Uygulamalar (*.exe *.com)", "Metin Dosyası (*.txt)"}); // Filtre ayarlama
    dosyaEkrani2.exec();

    if(!dosyaEkrani2.selectedFiles().isEmpty())
    {
        qDebug() << "Seçilen dosyanın yolu; " << dosyaEkrani2.selectedFiles().at(0).toStdString() << Qt::endl
        << "Seçilen dosyanın Url'si; " << dosyaEkrani2.selectedUrls().at(0) << Qt::endl;
    }
    else
    {
        qDebug() << "Herhangi bir dosya seçilmedi!" << Qt::endl;
    }

    QFileDialog klasorEkrani;
    klasorEkrani.setFileMode(QFileDialog::FileMode::Directory); // Klasör seçtirmek için
    klasorEkrani.exec();

    if(!klasorEkrani.selectedFiles().isEmpty())
    {
        qDebug() << "Seçilen dosyanın yolu; " << klasorEkrani.selectedFiles().at(0).toStdString() << Qt::endl
        << "Seçilen dosyanın Url'si; " << klasorEkrani.selectedUrls().at(0) << Qt::endl;
    }
    else
    {
        qDebug() << "Herhangi bir dosya seçilmedi!" << Qt::endl;
    }

    QFileDialog dosyaEkrani_Kaydetme;
    dosyaEkrani_Kaydetme.setAcceptMode(QFileDialog::AcceptMode::AcceptSave); // Kaydetme moduna ayarlama
    dosyaEkrani_Kaydetme.exec();

    if(!dosyaEkrani_Kaydetme.selectedFiles().isEmpty())
    {
        qDebug() << "Kayıt edilen dosyanın yolu; " << dosyaEkrani_Kaydetme.selectedFiles().at(0).toStdString() << Qt::endl
        << "Kayıt edilen dosyanın Url'si; " << dosyaEkrani_Kaydetme.selectedUrls().at(0) << Qt::endl;
    }
    else
    {
        qDebug() << "Herhangi bir dosya kayıt edilmedi!" << Qt::endl;
    }
}

mainwindow.h

#ifndef MAINWINDOW_H
#define MAINWINDOW_H

#include <QMainWindow>

QT_BEGIN_NAMESPACE
namespace Ui {
class MainWindow;
}

class MainWindow : public QMainWindow

void MainWindow::fileDialogGoster()
{
    QFileDialog dosyaEkrani; // Dosya seçim ekranı oluşturma
    dosyaEkrani.exec(); // Dosya seçim ekranını gösterme

    if(!dosyaEkrani.selectedFiles().isEmpty()) // Eğer kullanıcı dosya seçim ekranında bir dosya seçilirse
    {
        qDebug() << "Seçilen dosyanın yolu; " << dosyaEkrani.selectedFiles().at(0).toStdString() << Qt::endl
        << "Seçilen dosyanın Url'si; " << dosyaEkrani.selectedUrls().at(0) << Qt::endl;
    }
    else // Seçilmesze
    {
        qDebug() << "Herhangi bir dosya seçilmedi!" << Qt::endl;
    }

    QFileDialog dosyaEkrani2; // Dosya seçim ekranı oluşturma
    dosyaEkrani2.setNameFilters({"Fotoğraflar (*.png *.jpeg *.jpg)", "Uygulamalar (*.exe *.com)", "Metin Dosyası (*.txt)"}); // Filtre ayarlama
    dosyaEkrani2.exec();

    if(!dosyaEkrani2.selectedFiles().isEmpty())
    {
        qDebug() << "Seçilen dosyanın yolu; " << dosyaEkrani2.selectedFiles().at(0).toStdString() << Qt::endl
        << "Seçilen dosyanın Url'si; " << dosyaEkrani2.selectedUrls().at(0) << Qt::endl;
    }
    else
    {
        qDebug() << "Herhangi bir dosya seçilmedi!" << Qt::endl;
    }

    QFileDialog klasorEkrani;
    klasorEkrani.setFileMode(QFileDialog::FileMode::Directory); // Klasör seçtirmek için
    klasorEkrani.exec();

    if(!klasorEkrani.selectedFiles().isEmpty())
    {
        qDebug() << "Seçilen dosyanın yolu; " << klasorEkrani.selectedFiles().at(0).toStdString() << Qt::endl
        << "Seçilen dosyanın Url'si; " << klasorEkrani.selectedUrls().at(0) << Qt::endl;
    }
    else
    {
        qDebug() << "Herhangi bir dosya seçilmedi!" << Qt::endl;
    }

    QFileDialog dosyaEkrani_Kaydetme;
    dosyaEkrani_Kaydetme.setAcceptMode(QFileDialog::AcceptMode::AcceptSave); // Kaydetme moduna ayarlama
    dosyaEkrani_Kaydetme.exec();

    if(!dosyaEkrani_Kaydetme.selectedFiles().isEmpty())
    {
        qDebug() << "Kayıt edilen dosyanın yolu; " << dosyaEkrani_Kaydetme.selectedFiles().at(0).toStdString() << Qt::endl
        << "Kayıt edilen dosyanın Url'si; " << dosyaEkrani_Kaydetme.selectedUrls().at(0) << Qt::endl;
    }
    else
    {
        qDebug() << "Herhangi bir dosya kayıt edilmedi!" << Qt::endl;
    }
}

    << Qt::endl;
}
