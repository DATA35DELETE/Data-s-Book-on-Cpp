STL

STL Nedir?

STL yani Standart Template Library(Standart Şablon Kütüphanesi); C++ dilinde önceden bizler uğraşmayalım diye onaylanmış liste yapıları, algoritmalar ve yenilicilerle donatılmış bir C++ kütüphane birliğidir. Bileşenleri farklı farklı kütüphane dosyalarındadır.
Peki neden biz kendimiz tanımlamak yerine STL'de hazır olarak sunulan nesneleri kullanalım? Cevap hız! Evet kendimiz belkide daha iyi tanımlayabilir ama bu belki aylar belkide yıllarımız alırdı... Ama bizim yerimize yıllarca uğraşmış mühendisler tarafından tasarlanmış ve denetlenmiş olan bu sistemlerede haksızlık etmemek gerekir! Onlar gece gündüz çalıştılar ve bize bu muazzam kütüphaneyi armağan ettileri...

Containers

Containers yani kapsayıcılar, verileri aynı bir liste veya dizi gibi depolama ve organize etme işlevlerine sahip bir STL kütüphaneleridir.

Vector

Vector, "vector" kütüphanesinde yer alan veri ekleyip çıkartabildiğimiz, boyutu değişebilen bir liste türüdür. STL kütüphanelerinde kapsayıcılar sınıfına girer. Template ile beraber hangi veri türlerinin saklanacağı belirtilir.

#include <iostream>
#include <vector> // Vector kütüphanemiz

using namespace std;

int main()
{
    vector<int> ornekVector; // Bu şekilde tanımlanır
    ornekVector = {1,2,3,4}; // Bu şekilde de atanabilir

    int ornekVektordekiUcuncuIndexEleman = ornekVector.at(3); // Parantezlerindeki Index numarasına denk gelen elemanı alır
    cout << "ornekVector'deki 3. Index numarasina sahip deger: " << ornekVektordekiUcuncuIndexEleman << endl << endl;

    ornekVector.assign(3,12); // 0. Index'den nereye kadar hangi değer ile yazılacağı.
    // 4; 0. Index'den (n - 1)'den 2. Index'de kadarki alan. Diğer kısımları sil
    // 12; Neyin yazılacağı
    cout << "ornekVector'deki 2. Index numarasina sahip deger: " << ornekVector.at(2) << endl << endl;
    
    ornekVector = {10,20,30,40};

    int ornekVectorSonEleman = ornekVector.back(); // Vector'deki son elemanı alır
    cout << "ornekVector'deki son elemanin degeri: " << ornekVectorSonEleman << endl << endl;

    int ornekVectorIlkEleman = ornekVector.front(); // Vector'deki ilk elemanı alır
    cout << "ornekVector'deki ilk elemanin degeri: " << ornekVectorIlkEleman << endl << endl;
    
    int ornekVectorunKapasitesi = ornekVector.capacity(); // Vector'un kapasitesini alır
    cout << "ornekVector'un kapasitesinin degeri: " << ornekVectorunKapasitesi << endl << endl;

    ornekVector.clear(); // Vector'un içini temizler

    bool ornekVectorBosMu = ornekVector.empty(); // Vector'un boş olup olmadığına göre bir Bool değeri döndürür. Boş ise true değil ise false döndürür
    cout << "ornekVector bos mu: " << ornekVectorBosMu << endl;
    ornekVector = {11,22,33,44,55}; // ornekVector'e değerler ekledik
    ornekVectorBosMu = ornekVector.empty(); // Tekrardan kontrol ettik
    cout << "ornekVector bos mu: " << ornekVectorBosMu << endl << endl;

    int *ornekVectorunBellekAdresi = ornekVector.data(); // Vector'un bellek adresini alır
    cout << "ornekVector'un bellek adresi: " << ornekVectorunBellekAdresi << endl;
    int ornekVectorunBellekAdresindenIkinciIndexdekiEleman = ornekVectorunBellekAdresi[2]; // Bellek adresinden eleman alma, daha çok C++ programlama dilinden C programlama diline Vector aktarılırken kullanılır. Pointer adresleri bir nevi diziler için kullanıldığı ve Vector'lerde temelinde bir dizi oldukları için bu tarz bir yöntem bulunmaktadır ki C++ C köprüsü sağlıklı kurulsun. Ayrıca C'de Vecktor yoktur
    cout << "ornekVector'un bellek adresinden bulunan 2. Index numarasina sahip degeri: " << ornekVectorunBellekAdresindenIkinciIndexdekiEleman << endl << endl;

    ornekVector.emplace(ornekVector.begin(), 1); // Elemanların arasına eklemek için kullanılır.
    // ornekVector.begin(); Bir iteratör alır. Bir nevi STL'ler için "Hangi Index'den" gibi bir araçtır. Ayrıca sayı ekliyerek istediğimiz yere yazdırmasını da sağlayabiliriz
    // 1; Yazılacak değer
    cout << "ornekVector'un 0. Index numarasina sahip degeri; " << ornekVector.at(0) << endl << endl;

    ornekVector.emplace_back(66); // Vector'un en sonuna İteratör ihtiyaç duymadan direkt en sona ekleme yapar.
    cout << "ornekVector'deki elemanlar: " << endl;
    for (int eleman : ornekVector)
    {
        cout << eleman << endl;
    }
    cout << endl;

    ornekVector.erase(ornekVector.begin(), ornekVector.end() - 1); // Vector'lerde erase kullanımı İteratörler ile yapılır
    // ornekVector.begin(); Başlangıç İteratörü
    // ornekVector.end() - 1; Bitiş İteratörü
    cout << "ornekVector'deki elemanlar: " << endl;
    for (int eleman : ornekVector)
    {
        cout << eleman << endl;
    }
    cout << endl;

    ornekVector.insert(ornekVector.begin(), 77);
    cout << "ornekVector'deki elemanlar: " << endl;
    for (int eleman : ornekVector)
    {
        cout << eleman << endl;
    }
    cout << endl;

    unsigned long long ornekVectorunMaksBoyutu = ornekVector.max_size();
    cout << "ornekVector'un maksimum tutabilcegi degerler sayisi; " << ornekVectorunMaksBoyutu << endl << endl;

    ornekVector.push_back(88); // Sona bir eleman ekler
    cout << "ornekVector'deki elemanlar: " << endl;
    for (int eleman : ornekVector)
    {
        cout << eleman << endl;
    }
    cout << endl;

    ornekVector.pop_back(); // Sondan bir eleman siler
    cout << "ornekVector'deki elemanlar: " << endl;
    for (int eleman : ornekVector)
    {
        cout << eleman << endl;
    }
    cout << endl;

    ornekVector.reserve(100);
    cout << "ornekVector'un kapasitesi; " << ornekVector.capacity() << endl << endl;

    ornekVector.resize(10);
    int ornekVectorBoyutu = ornekVector.size();
    cout << "ornekVector'un boyutu; " << ornekVectorBoyutu << endl << endl;

    ornekVector.shrink_to_fit();
    cout << "ornekVector'un kapasitesi; " << ornekVector.capacity() << endl << endl;

    vector<int> baskaOrnekVector = {1,2,3,4,5,6};
    ornekVector.swap(baskaOrnekVector);
    cout << "ornekVector'deki elemanlar: " << endl;
    for (int eleman : ornekVector)
    {
        cout << eleman << endl;
    }
    cout << endl;

    cout << "baskaOrnekVector'deki elemanlar: " << endl;
    for (int eleman : baskaOrnekVector)
    {
        cout << eleman << endl;
    }
    cout << endl;

    return 0;
}

Stack

Stack yani yığın, diğer kapsayıcılara nazaran RAM'deki bölgelerden isminide aldığı Stack bölümünde verileri saklar. Mantığı ise LIFO denilen açılımı Last In First Out(Son Giren İlk Çıkar) prensibi ile kapsayıcı rolünde olan bir STL kütüphanesidir.
Peki neden FIFO var? Çünkü performans! FIFO veya Stack yapısı diğer STL yapılarına kıyasla daha küçük ve performanslıdır. Normal şartlarda aslında Stack bir yapı değil bir yöntemdir. C programlama dili geliştiricileri dinamik bir veri yapısı oluşturmayı çalışırken Stack yapısını keşfetmişlerdir. Stack, bir nevi yapıların bir birlerinin Pointerlarını tuttuğu ve gerektiğinde aralırında geçiş yapmayı olanak sağlayan bir yöntemdir. Yani Stack yapısının içerisindeki verileri kendisinden önce yazılan verilerin bellek adreslerini tutarlar. Geçiş yapılmak istendiğinde bellek adresleri aracığıyla geçiş mümkün kılınır. Bu yöntem daha çok soyut olarak işlenmektedir. Göreseli aşağıda verilmiştir.
Kendisini ve metodlarını inceleyelim;

#include <iostream>
#include <stack> // Kütüphanemiz

using namespace std;

int main()
{
    stack<int> ornekStack;

    bool ornekStackBosMu = ornekStack.empty();
    cout << "ornekStack bos mu; " << ornekStackBosMu << endl << endl;

    ornekStack.push(10); // Stack'de sadece push ile veri konulabilir
    ornekStack.push(20);
    ornekStack.push(30);
    ornekStack.push(40);

    //        |
    //        |
    //        |
    //        v

    // 40,30,20,10 sırası oluştu

    ornekStackBosMu = ornekStack.empty();
    cout << "ornekStack bos mu; " << ornekStackBosMu << endl << endl;

    int ornekStackBoyutu = ornekStack.size();
    cout << "ornekStack'in boyutu ; " << ornekStackBoyutu << endl << endl;

    ornekStack.pop(); // Stack'larde sadece pop ile veri silinebilir. En baştaki veriyi siler oda 40'a denk gelir

    ornekStackBoyutu = ornekStack.size();
    cout << "ornekStack'in boyutu ; " << ornekStackBoyutu << endl << endl;

    // Range tabanlı For Stack yapısında işlemez
    cout << "ornekStack'deki degerler;" << endl;
    int deger = 0;

    for (int i = 0; i < ornekStackBoyutu; i++)
    {
        deger = ornekStack.top(); // En üstdeki veriyi deger değişkenine ata

        cout << deger << endl;

        ornekStack.pop(); // Ardından en üstdeki veriyi sil
    }
    cout << endl;
    
    ornekStack.push(1);
    ornekStack.push(2);
    ornekStack.push(3);

    stack<int> baskaOrnekStack;
    baskaOrnekStack.push(11);
    baskaOrnekStack.push(22);
    baskaOrnekStack.push(33);

    ornekStack.swap(baskaOrnekStack);

    cout << "ornekStack'deki degerler;" << endl;
    deger = 0;

    for (int i = 0; i < ornekStackBoyutu; i++)
    {
        deger = ornekStack.top();

        cout << deger << endl;

        ornekStack.pop();
    }
    cout << endl;

    int baskaOrnekStackBoyutu = baskaOrnekStack.size();

    cout << "baskaOrnekStack'deki degerler;" << endl;
    deger = 0;

    for (int i = 0; i < baskaOrnekStackBoyutu; i++)
    {
        deger = baskaOrnekStack.top();

        cout << deger << endl;

        baskaOrnekStack.pop();
    }

    return 0;
}

Queue

Queue yani kuyruk, aynı Stack gibi RAM'in farklı kısmında depolanan ve performans açısından diğerlerinde daha iyi olan bir STL kapsayıcısıdır.
Stack'in keşfedeldiği dönemlerde Queue'de keşfedilmiştir. Mantığı ise FIFO yani First In First Out(İlk Giren İlk Çıkar) adı verilen bir metod ile yapılır. Yani sizin eklediğiniz her veri bir öncekinin arkasına yerleşir. Bu Stack'de tam tersidir. Örnek olarakda bir metro sistemini verebiliriz. Aynı hat üzerinde ilk sefere çıkan metro her zaman ilk önce son durağa ulaşan metrodur. Daha iyi anlamak için bir görsel üzerinden anlatalım;
Şimdi kodlarını inceleyelim;

#include <iostream>
#include <queue> // Kütüphanemiz

using namespace std;

int main()
{
    queue<int> ornekQueue; // Bu şekilde tanımlanır

    bool ornekQueueBosMu = ornekQueue.empty();
    cout << "ornekQueue bos mu; " << ornekQueueBosMu << endl;

    ornekQueue.push(10); // Queue'de veri eklemek için sadece Push kullanılır
    ornekQueue.push(20);
    ornekQueue.push(30);
    ornekQueue.push(40);

    //      |
    //      |
    //      |
    //      v
    // 10,20,30,40

    ornekQueueBosMu = ornekQueue.empty();
    cout << "ornekQueue bos mu; " << ornekQueueBosMu << endl << endl;

    int ornekQueueIlkEleman = ornekQueue.front();
    int ornekQueueSonEleman = ornekQueue.back();
    cout << "ornekQueue'nin ilk ve son elemani; " << ornekQueueIlkEleman << "," << ornekQueueSonEleman << endl << endl;

    int ornekQueueBoyutu = ornekQueue.size();
    cout << "ornekQueue'nin boyutu; " << ornekQueueBoyutu << endl;

    ornekQueueBoyutu = ornekQueue.size();
    cout << "ornekQueue'nin boyutu; " << ornekQueueBoyutu << endl << endl;

    // Range tabanlı For Queue yapısında işlemez
    cout << "ornekQueue'deki degerler;" << endl;
    int deger = 0;

    for (int i = 0; i < ornekQueueBoyutu; i++)
    {
        deger = ornekQueue.front(); // İlk giren veriyi ata

        ornekQueue.pop(); // Ardından en üstdeki veriyi sil
    }
    cout << endl;

    ornekQueue.push(11);
    ornekQueue.push(22);
    ornekQueue.push(33);

    queue<int> baskaOrnekQueue;

    baskaOrnekQueue.push(1);
    baskaOrnekQueue.push(2);
    baskaOrnekQueue.push(3);

    int baskaOrnekQueueBoyutu = baskaOrnekQueue.size();
    cout << "baskaOrnekQueue'nin boyutu; " << baskaOrnekQueueBoyutu << endl << endl;

    ornekQueue.swap(baskaOrnekQueue);

    cout << "ornekQueue'deki degerler;" << endl;
    deger = 0;

    for (int i = 0; i < ornekQueueBoyutu; i++)
    {
        deger = ornekQueue.front();

        cout << deger << endl;

        ornekQueue.pop();
    }
    cout << endl;

    cout << "baskaOrnekQueue'deki degerler;" << endl;
    deger = 0;

    for (int i = 0; i < baskaOrnekQueueBoyutu; i++)
    {
        deger = baskaOrnekQueue.front();

        cout << deger << endl;

        baskaOrnekQueue.pop();
    }

Priority Queue

Priority Queue, Queue kütüphanesinin altında bulunan ve Queue gibi olan fakat girilen değerleri alfanumarik bir şekilde büyükten küçüğe sıralar sıralar. Bir diğer özelliği ise RAM'in Queue gibi Stack bölümünden ziyade Heap bölümünde yer alır. "Bellek Yönetimi" sayfasında daha detaylı işleceğiz fakat şuanlık Priority Queue'nin RAM'in Heap bölümünde depolandığını bilelim.

Kodlarını inceleyelim;

#include <iostream>
#include <queue> // Priority Queue kütüphanesi Queue kütüphanesiyle beraber gelir

using namespace std;

int main()
{
    priority_queue<int> ornekPriority_Queue; // Bu şekilde tanımlanır

    bool ornekPriority_QueueBosMu = ornekPriority_Queue.empty();
    cout << "ornekPriority_Queue bos mu; " << ornekPriority_QueueBosMu << endl;

    ornekPriority_Queue.push(50);
    ornekPriority_Queue.push(20);
    ornekPriority_Queue.push(10);
    ornekPriority_Queue.push(40);
    ornekPriority_Queue.push(30);

    //          |
    //          |
    //          |
    //          v

    // 50,40,30,20,10; Rastgele push'lasakda Priority Queue onu otomatik olarak büyükten küçüğe sıralar

    ornekPriority_QueueBosMu = ornekPriority_Queue.empty();
    cout << "ornekPriority_Queue bos mu; " << ornekPriority_QueueBosMu << endl << endl;

    int ornekPriority_QueueBoyutu = ornekPriority_Queue.size();
    cout << "ornekPriority_Queue'nin boyutu; " << ornekPriority_QueueBoyutu << endl;

    ornekPriority_Queue.pop(); // Sıralama sonucu en üstte 50 değeri bulunduğu için o listeden çıkartıldı

    ornekPriority_QueueBoyutu = ornekPriority_Queue.size();
    cout << "ornekPriority_Queue'nin boyutu; " << ornekPriority_QueueBoyutu << endl << endl;

    // Range tabanlı For Priority Queue yapısında işlemez
    cout << "ornekPriority_Queue'deki degerler;" << endl;
    int deger = 0;

    for (int i = 0; i < ornekPriority_QueueBoyutu; i++)
    {
        deger = ornekPriority_Queue.top(); // En üstdeki veriyi deger değişkenine ata. Top kullanılır çünkü Priority Queue'de Queue gibi front kullanılırmaz. Çünkü sıralama direkt olmaz fonksiyonlar aracığıyla karşılaştırılarak yapılır

        cout << deger << endl;

        ornekPriority_Queue.pop(); // Ardından veriyi sil
    }
    cout << endl;

    ornekPriority_Queue.push(3);
    ornekPriority_Queue.push(1);
    ornekPriority_Queue.push(4);
    ornekPriority_Queue.push(2);

    priority_queue<int> baskaOrnekPriority_Queue;

    baskaOrnekPriority_Queue.push(22);
    baskaOrnekPriority_Queue.push(44);
    baskaOrnekPriority_Queue.push(11);
    baskaOrnekPriority_Queue.push(33);

    int baskaOrnekPriority_QueueBoyutu = baskaOrnekPriority_Queue.size();
    cout << "baskaOrnekPriority_Queue'nin boyutu; " << baskaOrnekPriority_QueueBoyutu << endl << endl;

    ornekPriority_Queue.swap(baskaOrnekPriority_Queue);

    cout << "ornekPriority_Queue'deki degerler;" << endl;
    deger = 0;

    for (int i = 0; i < ornekPriority_QueueBoyutu; i++)
    {
        deger = ornekPriority_Queue.top();

        cout << deger << endl;

        ornekPriority_Queue.pop();
    }
    cout << endl;

    cout << "baskaOrnekPriority_Queue'deki degerler;" << endl;
    deger = 0;

    for (int i = 0; i < baskaOrnekPriority_QueueBoyutu; i++)
    {
        deger = baskaOrnekPriority_Queue.top(); // En üstdeki veriyi deger değişkenine ata. Top kullanılır çünkü Priority Queue'de Queue gibi front kullanılırmaz. Çünkü sıralama direkt olmaz fonksiyonlar aracığıyla karşılaştırılarak yapılır

        cout << deger << endl;

        baskaOrnekPriority_Queue.pop(); // Ardından veriyi sil
    }
    cout << endl;
}

Set

Set, içerisindeki verileri küçükten büyüğe sıralar. Ayrıca girilen verilerin içerisindeki farklı olmasına dikkat eder ve aynı verileri kabul etmez.
            
#include <iostream>
#include <set> // Kütüphanemiz

using namespace std;

int main()
{
    set<int> ornekSet; // Bu şekilde tanımlanır

    bool ornekSetBosMu = ornekSet.empty();
    cout << "ornekSet bos mu; " << ornekSetBosMu << endl;

    ornekSet = {50,20,10,10,20,30,40,50}; // Bu şekilde de veri atanır

    ornekSetBosMu = ornekSet.empty();
    cout << "ornekSet bos mu; " << ornekSetBosMu << endl << endl;

    cout << "ornekSet'deki elemanlar;" << endl;
    for (int eleman : ornekSet)
    {
        cout << eleman << endl;
    }
    cout << endl;

    bool ornekSetdeyirmiDegeriVarMi = ornekSet.contains(20); // Set içerisinde parantezlerdeki değeri aramamızı sağlar. Bulursa true bulamazsa false döner
    cout << "ornekSet'in icerisinde 20 degeri var mi; " << ornekSetdeyirmiDegeriVarMi << endl << endl;

    ornekSet.emplace(60); // Emplace, normal şartlarda bizden bir İteratör ister fakat Set otomatik sıraladığı için emplace'de Iteratör kullanılmaz. Set gerisini halleder
    ornekSet.emplace(60); // Aynı değeri 2 kere eklemeyi deniyelim

    cout << "ornekSet'deki elemanlar;" << endl;
    for (int eleman : ornekSet)
    {
        cout << eleman << endl;
    }
    cout << endl;

    ornekSet.erase(10); // Girilen değeri Set'den siler

    cout << "ornekSet'deki elemanlar;" << endl;
    for (int eleman : ornekSet)
    {
        cout << eleman << endl;
    }
    cout << endl;

    ornekSet.insert(65); // Parantezlerindeki veriyi Set'e ekler

    cout << "ornekSet'deki elemanlar;" << endl;
    for (int eleman : ornekSet)
    {
        cout << eleman << endl;
    }
    cout << endl;

    unsigned long long ornekSetSetMaksBoyutu = ornekSet.max_size();
    cout <<"ornekSet'in maksimum boyutu; " << ornekSetSetMaksBoyutu << endl << endl;

    set<int> baskaOrnekSet = {11,22,33,44,55};

    ornekSet.merge(baskaOrnekSet);

    cout << "ornekSet'deki elemanlar;" << endl;
    for (int eleman : ornekSet)
    {
        cout << eleman << endl;
    }
    cout << endl;

    int ornekSetBoyutu = ornekSet.size();
    cout << "ornekSet'in boyutu; " << ornekSetBoyutu << endl << endl;

    set<int> baskaOrnekSet2 = {1,2,3,4,5,6};

    ornekSet.swap(baskaOrnekSet2);

    cout << "ornekSet'deki elemanlar;" << endl;
    for (int eleman : ornekSet)
    {
        cout << eleman << endl;
    }
    cout << endl;

    cout << "baskaOrnekSet2'deki elemanlar;" << endl;
    for (int eleman : baskaOrnekSet2)
    {
        cout << eleman << endl;
    }
    cout << endl;

    return 0;
}
