STL

STL Nedir?

STL yani Standart Template Library(Standart Şablon Kütüphanesi); C++ dilinde önceden bizler uğraşmayalım diye onaylanmış liste yapıları, algoritmalar ve yenilicilerle donatılmış bir C++ kütüphane birliğidir. Bileşenleri farklı farklı kütüphane dosyalarındadır.
Peki neden biz kendimiz tanımlamak yerine STL'de hazır olarak sunulan nesneleri kullanalım? Cevap hız! Evet kendimiz belkide daha iyi tanımlayabilir ama bu belki aylar belkide yıllarımız alırdı... Ama bizim yerimize yıllarca uğraşmış mühendisler tarafından tasarlanmış ve denetlenmiş olan bu sistemlerede haksızlık etmemek gerekir! Onlar gece gündüz çalıştılar ve bize bu muazzam kütüphaneyi armağan ettileri...

Containers

Containers yani kapsayıcılar, verileri aynı bir liste veya dizi gibi depolama ve organize etme işlevlerine sahip bir STL kütüphaneleridir.

Vector

Vector, "vector" kütüphanesinde yer alan veri ekleyip çıkartabildiğimiz, boyutu değişebilen bir liste türüdür. STL kütüphanelerinde kapsayıcılar sınıfına girer. Template ile beraber hangi veri türlerinin saklanacağı belirtilir.

#include <iostream>
#include <vector> // Vector kütüphanemiz

using namespace std;

int main()
{
    vector<int> ornekVector; // Bu şekilde tanımlanır
    ornekVector = {1,2,3,4}; // Bu şekilde de atanabilir

    int ornekVektordekiUcuncuIndexEleman = ornekVector.at(3); // Parantezlerindeki Index numarasına denk gelen elemanı alır
    cout << "ornekVector'deki 3. Index numarasina sahip deger: " << ornekVektordekiUcuncuIndexEleman << endl << endl;

    ornekVector.assign(3,12); // 0. Index'den nereye kadar hangi değer ile yazılacağı.
    // 4; 0. Index'den (n - 1)'den 2. Index'de kadarki alan. Diğer kısımları sil
    // 12; Neyin yazılacağı
    cout << "ornekVector'deki 2. Index numarasina sahip deger: " << ornekVector.at(2) << endl << endl;
    
    ornekVector = {10,20,30,40};

    int ornekVectorSonEleman = ornekVector.back(); // Vector'deki son elemanı alır
    cout << "ornekVector'deki son elemanin degeri: " << ornekVectorSonEleman << endl << endl;

    int ornekVectorIlkEleman = ornekVector.front(); // Vector'deki ilk elemanı alır
    cout << "ornekVector'deki ilk elemanin degeri: " << ornekVectorIlkEleman << endl << endl;
    
    int ornekVectorunKapasitesi = ornekVector.capacity(); // Vector'un kapasitesini alır
    cout << "ornekVector'un kapasitesinin degeri: " << ornekVectorunKapasitesi << endl << endl;

    ornekVector.clear(); // Vector'un içini temizler

    bool ornekVectorBosMu = ornekVector.empty(); // Vector'un boş olup olmadığına göre bir Bool değeri döndürür. Boş ise true değil ise false döndürür
    cout << "ornekVector bos mu: " << ornekVectorBosMu << endl;
    ornekVector = {11,22,33,44,55}; // ornekVector'e değerler ekledik
    ornekVectorBosMu = ornekVector.empty(); // Tekrardan kontrol ettik
    cout << "ornekVector bos mu: " << ornekVectorBosMu << endl << endl;

    int *ornekVectorunBellekAdresi = ornekVector.data(); // Vector'un bellek adresini alır
    cout << "ornekVector'un bellek adresi: " << ornekVectorunBellekAdresi << endl;
    int ornekVectorunBellekAdresindenIkinciIndexdekiEleman = ornekVectorunBellekAdresi[2]; // Bellek adresinden eleman alma, daha çok C++ programlama dilinden C programlama diline Vector aktarılırken kullanılır. Pointer adresleri bir nevi diziler için kullanıldığı ve Vector'lerde temelinde bir dizi oldukları için bu tarz bir yöntem bulunmaktadır ki C++ C köprüsü sağlıklı kurulsun. Ayrıca C'de Vecktor yoktur
    cout << "ornekVector'un bellek adresinden bulunan 2. Index numarasina sahip degeri: " << ornekVectorunBellekAdresindenIkinciIndexdekiEleman << endl << endl;

    ornekVector.emplace(ornekVector.begin(), 1); // Elemanların arasına eklemek için kullanılır.
    // ornekVector.begin(); Bir iteratör alır. Bir nevi STL'ler için "Hangi Index'den" gibi bir araçtır. Ayrıca sayı ekliyerek istediğimiz yere yazdırmasını da sağlayabiliriz
    // 1; Yazılacak değer
    cout << "ornekVector'un 0. Index numarasina sahip degeri; " << ornekVector.at(0) << endl << endl;

    ornekVector.emplace_back(66); // Vector'un en sonuna İteratör ihtiyaç duymadan direkt en sona ekleme yapar.
    cout << "ornekVector'deki elemanlar: " << endl;
    for (int eleman : ornekVector)
    {
        cout << eleman << endl;
    }
    cout << endl;

    ornekVector.erase(ornekVector.begin(), ornekVector.end() - 1); // Vector'lerde erase kullanımı İteratörler ile yapılır
    // ornekVector.begin(); Başlangıç İteratörü
    // ornekVector.end() - 1; Bitiş İteratörü
    cout << "ornekVector'deki elemanlar: " << endl;
    for (int eleman : ornekVector)
    {
        cout << eleman << endl;
    }
    cout << endl;

    ornekVector.insert(ornekVector.begin(), 77);
    cout << "ornekVector'deki elemanlar: " << endl;
    for (int eleman : ornekVector)
    {
        cout << eleman << endl;
    }
    cout << endl;

    unsigned long long ornekVectorunMaksBoyutu = ornekVector.max_size();
    cout << "ornekVector'un maksimum tutabilcegi degerler sayisi; " << ornekVectorunMaksBoyutu << endl << endl;

    ornekVector.push_back(88); // Sona bir eleman ekler
    cout << "ornekVector'deki elemanlar: " << endl;
    for (int eleman : ornekVector)
    {
        cout << eleman << endl;
    }
    cout << endl;

    ornekVector.pop_back(); // Sondan bir eleman siler
    cout << "ornekVector'deki elemanlar: " << endl;
    for (int eleman : ornekVector)
    {
        cout << eleman << endl;
    }
    cout << endl;

    ornekVector.reserve(100);
    cout << "ornekVector'un kapasitesi; " << ornekVector.capacity() << endl << endl;

    ornekVector.resize(10);
    int ornekVectorBoyutu = ornekVector.size();
    cout << "ornekVector'un boyutu; " << ornekVectorBoyutu << endl << endl;

    ornekVector.shrink_to_fit();
    cout << "ornekVector'un kapasitesi; " << ornekVector.capacity() << endl << endl;

    vector<int> baskaOrnekVector = {1,2,3,4,5,6};
    ornekVector.swap(baskaOrnekVector);
    cout << "ornekVector'deki elemanlar: " << endl;
    for (int eleman : ornekVector)
    {
        cout << eleman << endl;
    }
    cout << endl;

    cout << "baskaOrnekVector'deki elemanlar: " << endl;
    for (int eleman : baskaOrnekVector)
    {
        cout << eleman << endl;
    }
    cout << endl;

    return 0;
}

Stack

Stack yani yığın, diğer kapsayıcılara nazaran RAM'deki bölgelerden isminide aldığı Stack bölümünde verileri saklar. Mantığı ise LIFO denilen açılımı Last In First Out(Son Giren İlk Çıkar) prensibi ile kapsayıcı rolünde olan bir STL kütüphanesidir.
Peki neden FIFO var? Çünkü performans! FIFO veya Stack yapısı diğer STL yapılarına kıyasla daha küçük ve performanslıdır. Normal şartlarda aslında Stack bir yapı değil bir yöntemdir. C programlama dili geliştiricileri dinamik bir veri yapısı oluşturmayı çalışırken Stack yapısını keşfetmişlerdir. Stack, bir nevi yapıların bir birlerinin Pointerlarını tuttuğu ve gerektiğinde aralırında geçiş yapmayı olanak sağlayan bir yöntemdir. Yani Stack yapısının içerisindeki verileri kendisinden önce yazılan verilerin bellek adreslerini tutarlar. Geçiş yapılmak istendiğinde bellek adresleri aracığıyla geçiş mümkün kılınır. Bu yöntem daha çok soyut olarak işlenmektedir. Göreseli aşağıda verilmiştir.
Kendisini ve metodlarını inceleyelim;

#include <iostream>
#include <stack> // Kütüphanemiz

using namespace std;

int main()
{
    stack<int> ornekStack;

    bool ornekStackBosMu = ornekStack.empty();
    cout << "ornekStack bos mu; " << ornekStackBosMu << endl << endl;

    ornekStack.push(10); // Stack'de sadece push ile veri konulabilir
    ornekStack.push(20);
    ornekStack.push(30);
    ornekStack.push(40);

    //        |
    //        |
    //        |
    //        v

    // 40,30,20,10 sırası oluştu

    ornekStackBosMu = ornekStack.empty();
    cout << "ornekStack bos mu; " << ornekStackBosMu << endl << endl;

    int ornekStackBoyutu = ornekStack.size();
    cout << "ornekStack'in boyutu ; " << ornekStackBoyutu << endl << endl;

    ornekStack.pop(); // Stack'larde sadece pop ile veri silinebilir. En baştaki veriyi siler oda 40'a denk gelir

    ornekStackBoyutu = ornekStack.size();
    cout << "ornekStack'in boyutu ; " << ornekStackBoyutu << endl << endl;

    // Range tabanlı For Stack yapısında işlemez
    cout << "ornekStack'deki degerler;" << endl;
    int deger = 0;

    for (int i = 0; i < ornekStackBoyutu; i++)
    {
        deger = ornekStack.top(); // En üstdeki veriyi deger değişkenine ata

        cout << deger << endl;

        ornekStack.pop(); // Ardından en üstdeki veriyi sil
    }
    cout << endl;
    
    ornekStack.push(1);
    ornekStack.push(2);
    ornekStack.push(3);

    stack<int> baskaOrnekStack;
    baskaOrnekStack.push(11);
    baskaOrnekStack.push(22);
    baskaOrnekStack.push(33);

    ornekStack.swap(baskaOrnekStack);

    cout << "ornekStack'deki degerler;" << endl;
    deger = 0;

    for (int i = 0; i < ornekStackBoyutu; i++)
    {
        deger = ornekStack.top();

        cout << deger << endl;

        ornekStack.pop();
    }
    cout << endl;

    int baskaOrnekStackBoyutu = baskaOrnekStack.size();

    cout << "baskaOrnekStack'deki degerler;" << endl;
    deger = 0;

    for (int i = 0; i < baskaOrnekStackBoyutu; i++)
    {
        deger = baskaOrnekStack.top();

        cout << deger << endl;

        baskaOrnekStack.pop();
    }

    return 0;
}

Queue

Queue yani kuyruk, aynı Stack gibi RAM'in farklı kısmında depolanan ve performans açısından diğerlerinde daha iyi olan bir STL kapsayıcısıdır.
Stack'in keşfedeldiği dönemlerde Queue'de keşfedilmiştir. Mantığı ise FIFO yani First In First Out(İlk Giren İlk Çıkar) adı verilen bir metod ile yapılır. Yani sizin eklediğiniz her veri bir öncekinin arkasına yerleşir. Bu Stack'de tam tersidir. Örnek olarakda bir metro sistemini verebiliriz. Aynı hat üzerinde ilk sefere çıkan metro her zaman ilk önce son durağa ulaşan metrodur. Daha iyi anlamak için bir görsel üzerinden anlatalım;
Şimdi kodlarını inceleyelim;

#include <iostream>
#include <queue> // Kütüphanemiz

using namespace std;

int main()
{
    queue<int> ornekQueue; // Bu şekilde tanımlanır

    bool ornekQueueBosMu = ornekQueue.empty();
    cout << "ornekQueue bos mu; " << ornekQueueBosMu << endl;

    ornekQueue.push(10); // Queue'de veri eklemek için sadece Push kullanılır
    ornekQueue.push(20);
    ornekQueue.push(30);
    ornekQueue.push(40);

    //      |
    //      |
    //      |
    //      v
    // 10,20,30,40

    ornekQueueBosMu = ornekQueue.empty();
    cout << "ornekQueue bos mu; " << ornekQueueBosMu << endl << endl;

    int ornekQueueIlkEleman = ornekQueue.front();
    int ornekQueueSonEleman = ornekQueue.back();
    cout << "ornekQueue'nin ilk ve son elemani; " << ornekQueueIlkEleman << "," << ornekQueueSonEleman << endl << endl;

    int ornekQueueBoyutu = ornekQueue.size();
    cout << "ornekQueue'nin boyutu; " << ornekQueueBoyutu << endl;

Priority Queue

Priority Queue, Queue kütüphanesinin altında bulunan ve Queue gibi olan fakat girilen değerleri alfanumarik bir şekilde büyükten küçüğe sıralar sıralar. Bir diğer özelliği ise RAM'in Queue gibi Stack bölümünden ziyade Heap bölümünde yer alır. "Bellek Yönetimi" sayfasında daha detaylı işleceğiz fakat şuanlık Priority Queue'nin RAM'in Heap bölümünde depolandığını bilelim.

Kodlarını inceleyelim;

Map

Map, diğer kapsayıcalardan farklı olarak anahtar-değer ilişkisi ile değer tutar. Temlate'ine iki adet tür alır. İlki anahtar değerin türü diğeri ise değerin veri türüdür.
Peki nedir bu anahtar-değer nam-ı diğer Key-Value ilişikisi? Mantıkları ise bir anahtar ve kilitli kutuya benzer... Anahtarımızla kilitli kutuyu açarız ve içerisindeki tanımlanmış veriyi elde ederiz. Fakat başka anahtarla erişemeyiz. İşte Map böyle bir yapıdır. Ayrıca Map'de Index sıralaması yoktur. Fakat kendi içinde Anahtar değerlerin alfanumarik bir şekilde sıralar.

Şimdi gelin kodlarını inceleyelim;

{

    ornekMap = {

    cout << "C programlama dili su tarihte cikmistir; " << CNinCiktigiTarih << endl << endl;

İteratörler(Yeniliciler)

Yeniliyiciler, STL kapsayacıların elemanları arasında geçiş yapmamız sağlayan, içeriğini kullanabilmemizi sağlayan sınıflardır. Her kapsayıcıda bu özellik bulunur.
Tüm kapsayıcılarda bulunsada ben yenilicileri Vector üzerinden anlatıcağım;

#include <iostream>
#include <vector>

using namespace std;

int main()
{
    vector<int> raklamlar = {1,2,3,4,5,6,7,8,9,0}; // Rakamlarımız

    cout << "rakamlar Vector'nun elemanlari;" << endl;
    for (auto rakam = raklamlar.begin(); rakam != raklamlar.end(); rakam++)
    {
        cout << rakam.base() << " -> " << *rakam << endl;  
    }
    cout << endl;
    // Bu For döngüsü yenilicilerler rakamlar'daki elemanların bellek adresleriyle birlikte değerlerini yazdırdık. Peki nasıl?
    // Olay şu: "auto rakam = raklamlar.begin();" dediğimiz kısım bizim için bir nevi başlangıç Index değeri gibidir. Başlangıç olarak .begin yani "Bana elemanı alcak şekilde yenilicisini ver" diyoruz. Fonksiyonda bizlere ilk elemanın yenilicisini veriyor. Şimdi yenilicilere demiştik ki içeriğini dokunulmadan kapsayıcıların elemanlarını sıralamamızı sağlayan yapılar demiştik. Begin fonksiyonu aslında bize ilk elemanın bellek adresini vermiş oldu. Ardından "auto" ile hangi tür olduğu fark etmekmizin rakam değişkenine atadı
    // Daha sonra kısımın "rakam != raklamlar.end();" anlamı ise rakam değişkeni, yenilicisi veya işaretçisi rakamlar Vector'unun son elemanın yenilicisine veya bellek adresine eşit olmadığı süreççe dön. Böylelikle For döndügüsü ilk yeniliciden son yeniliciye yani ilk elemanın bellek adresinden son elemanın bellek adresine kadarki bütün elemanları işleme alıcaktır
    // Son kısım olan "rakam++" ise Pointer Aritmatiğine dayanır. "Pointerlar" sayfasında öğrendiğimiz üzere bir işaretçiye bir değeri eklersek bellek adresindeki hemen yanındaki bellek adresini işaret etmeye başlar. Böylelikle diziler veya bütünleşik yapılar oluşmaktır. Aynı mantığı döngü her döndüğünde bir sonraki bellek adresine geçiriyoruz ki elemanlar arasında geçiş yapalım
    // Son olarak yenilicinin tuttuğu bellek adresini base fonksiyonu alıp, yenilicinin tuttuğu değeride bellek adresinden çekip ekrana yazdırıyoruz

    cout << "rakamlar Vector'nun elemanlari;" << endl;
    for (auto rakam = raklamlar.cbegin(); rakam != raklamlar.cend(); rakam++)
    {
        cout << rakam.base() << " -> " << *rakam << endl;  
    }
    cout << endl;
    // cbegin ve cend'in begin ve end'den farkı ise "sabit yeniliyici" döndürmesidir. Ancak kayda değer bir farklılık göstermezler

    cout << "rakamlar Vector'nun tersten elemanlari;" << endl;
    for (auto rakam = raklamlar.rbegin(); rakam != raklamlar.rend(); rakam++)
    {
        cout << *rakam << endl;  
    }
    cout << endl;
    // rbegin ve rend begin ve end'in terslerini verir. Yani begin = rend, end = rbegin

    return 0;
}

Çıktı;

Algoritmalar

Algoritmalar, kapsayacıların içeriğini sıralama, sayma veya içlerinden değer bulma gibi işlemler için yazılmış fonksiyonlardır. Genel olarak tüm kapsayıcılarda kullanılabilir.
Örnek olması açısından ben bu fonksiyonları Vector üzerinden yazdım;

#include <iostream>
#include <vector>
#include <algorithm> // STL algoritmalarının bulunduğu kütüphanedir
#include <numeric> // accumulate fonksiyonu için gerekli kütüphane

using namespace std;

int main()
{
    vector<int> sayilar = {7,5,8,3,1,10,4,6,2,9}; // Örnek Vector'umuz

    cout << "sayilar Vector'unun elemanlari;" << endl;
    for (int sayi : sayilar)
    {
        cout << sayi << endl;
    }
    cout << endl;
    
    sort(sayilar.begin(), sayilar.end()); // Girilen başlangıç ve bitiş yenilicilerini alfanumarik olarak küçükten büyüğe sıralar. Eğer ters yenileyicisi verilirse(rbegin ve rend) alfonumarik olarak büyükten küçüğe sıralar
    
    cout << "sayilar Vector'unun elemanlari;" << endl;
    for (int sayi : sayilar)
    {
        cout << sayi << endl;
    }
    cout << endl;

    auto eleman7 = find(sayilar.begin(), sayilar.end(), 7); // Girilen başlangıç ve bitiş yenilicilerinin içerisinde 3. parametre olan 7 sayısının yenilicisini yani bellek adresini verir
    cout << "sayilar Vector'unun 7 degerine sahip elemanin bellek adresi ve degeri; " << eleman7.base() << " , " << *eleman7 << endl << endl;

    cout << "sayilar Vector'unun elemanlari;" << endl;
    for_each(sayilar.begin(), sayilar.end(),
    [] (auto sayi) 
    {
        cout << sayi << endl;
    }
    ); // Girilen başlangıç ve bitiş yenilicilerini 3. parametredeki fonksiyona veya lambda fonksiyonuna ilk parametre olarak başlangıçtan bitişe doğru değerleri gönderir. Tüm elemanları vericek şekilde döndürür
    cout << endl;

    vector<int> sayilar2;
    sayilar2.resize(sayilar.size());

    transform(sayilar.begin(), sayilar.end(), sayilar2.begin(), 
    [] (auto sayi)
    {
        return sayi;
    }
    ); // Girilen başlangıç ve bitiş yenilicilerini 3. parametredeki başka bir kapsayıcının başlangıç yenileyicisine 4. parametredeki lambda veya fonksiyona ilk parametre olarak başlangıçtan bitiş yenileyicisine sırayla gönderir. Lambdan veya fonksiyon return'lenen değer 3. parametredeki yenileyiciye aktarılır

    cout << "sayilar2 Vector'unun elemanlari;" << endl;
    for (int sayi : sayilar2)
    {
        cout << sayi << endl;
    }
    cout << endl;

    sayilar.push_back(10);
    sayilar.push_back(10);
    sayilar.push_back(10);
    sayilar.push_back(10);
    sayilar.push_back(10);

    int sayilardaKacAdet10Var = count(sayilar.begin(), sayilar.end(), 10); // Girilen başlangıç ve bitiş yenilicilerinin arasında kaç adet 3. parametredeki değerin olduğu hesaplar ve bir Integer değeri döner

    cout << "sayilar Vector'unde kac adet 10 degeri var; " << sayilardaKacAdet10Var << endl;

    sayilardaKacAdet10Var = count_if(sayilar.begin(), sayilar.end(),
    [] (auto sayi)
    {
        if (sayi == 10)
            return 1;
        return 0;
    }
    ); // count'un lambda veya fonksiyonlu halidir. Lambdan veya fonksiyondan dönen değer 0'dan büyük ise bir, 0 ise herhangi bir ekleme yapmaz.
    cout << "sayilar Vector'unde kac adet 10 degeri var; " << sayilardaKacAdet10Var << endl << endl;

    int sayilarElemanlarininToplami = accumulate(sayilar.begin(), sayilar.end(), 0); //  // Girilen başlangıç ve bitiş yenilicilerinin arasında değerleri 3. parametredeki başlangıç değer olarak toplar ve bir Integer değeri döndürür

    cout << "sayilar elemanlarinin toplami; " << sayilarElemanlarininToplami << endl << endl;

    return 0;
}

Pair

Pair veya çiftli, aynı anda iki tane veri tutabilen ve first ve second olara ulaşabilceğimiz bir sınıftır. Utility kütüphanesinde yer alır.
Bir insan hem ismini hemde soyismini sanki tek bir veriymiş gibi beyninde saklar. Fakat gerektiğinde rahatlıkla ayırabilir çünkü ismimizi ve soyismimizi ayrı şekilde kullanmamız gereken zamanlar olur. İşte gelin bu yapıyı C++ programlama dilinde yazalım;

#include <iostream>
#include <utility> // Kütüphanemiz

using namespace std;

int main()
{
    pair<string, string> insan1 = {"Data", "Delete"};

    cout << "insan1'in ismi ve soyismi; " << insan1.first << " " << insan1.second << endl << endl;
}

Tuple

Tuple, Pair'in ikiden fazla tanımlamayı izin veren halidir. first ve second olarak erişilmez fakat Template içerisinde yazılan Index numarası ve get fonksiyonu ile değere ulaşılır. Tuple kütüphanesinde yer alır.
insan1'e birde yaş ekliyelim bakalım;

#include <iostream>
#include <tuple> // Kütüphanemiz

using namespace std;

int main()
{
    tuple<string, string, int> insan1 = {"Data", "Delete", 16};

    cout << "insan1'in ismi, soyismi ve yasi; " << get<0>(insan1) << " " << get<1>(insan1) << " " << get<2>(insan1) << endl << endl;
}

Çıktısı ise;

Optional

Optional veya opsiyonel, tek bir veri türünün tek bir nesnede işlem yapmamızı sağlayan bir sınıftır. Optional kütüphanesinde yer alır. C++17 ile standartlaşmıştır.
Hadi inceleyelim;

#include <iostream>
#include <optional> // Kütüphanemiz

using namespace std;

int main()
{
    optional<int> ornekOptional; // Bu şekilde tanımlanır

    bool ornekOptionalBosMu = ornekOptional.has_value(); // Boş mu değil mi diye kontrol eder
    cout << "ornekOptional bos mu; " << ornekOptionalBosMu << endl;

    ornekOptional.emplace(100); // Atama yapar

    ornekOptionalBosMu = ornekOptional.has_value();
    cout << "ornekOptional bos mu; " << ornekOptionalBosMu << endl << endl;

    int ornekOptionalNinDegeri = ornekOptional.value(); // Optional'in değerini alır
    cout << "ornekOptional degeri; " << ornekOptionalNinDegeri << endl;

    ornekOptional.reset(); // Değeri nullptr'e çevirir

    ornekOptional.emplace(200);

    ornekOptionalNinDegeri = ornekOptional.value();
    cout << "ornekOptional degeri; " << ornekOptionalNinDegeri << endl << endl;

    ornekOptional.reset();

    ornekOptionalNinDegeri = ornekOptional.value_or(0); // Eğer nullptr ise parantezlerdeki değer döndür
    cout << "ornekOptional degeri; " << ornekOptionalNinDegeri << endl << endl;

    ornekOptional.emplace(10);

    optional<int> baskaOrnekOptional = 20;

    ornekOptional.swap(baskaOrnekOptional);

    ornekOptionalNinDegeri = ornekOptional.value();
    cout << "ornekOptional degeri; " << ornekOptionalNinDegeri << endl;

    int baskaOrnekOptionalNinDegeri = baskaOrnekOptional.value();
    cout << "baskaOrnekOptional degeri; " << baskaOrnekOptionalNinDegeri << endl;

    return 0;
}

Çıktısı;

Variant

Variant, aynı Union gibi tek bir değer fakat birden fazla türden depolama sağlayabilen bir yapıdır. C++17 ile standartlaşmıştır. Variant kütüphanesinde bulunur.

Hadi inceleyelim;

#include <iostream>
#include <variant> // Kütüphanemiz

using namespace std;

int main()
{
    variant<int, float> intVeFloat; // Bu şekilde tanımlıyoruz

    intVeFloat.emplace<int>(1234); // Emplace'in Template'ine hangi tür girildi ise o türe göre değer atanır

    int kullanilanIndex = intVeFloat.index(); // Hangi veri türü kullanılıyorsa onun Index'ini almak için
    cout << "intVeFloat'in kullanilan Index numarasi; " << kullanilanIndex << endl << endl;

    int intVeFloatIntDegeri = get<0>(intVeFloat);
    cout << "intVeFloat Variant'nin 0. Index'deki turunun degeri; " << intVeFloatIntDegeri << endl << endl;

    intVeFloat.emplace<float>(12.54f);

    kullanilanIndex = intVeFloat.index();
    cout << "intVeFloat'in kullanilan Index numarasi; " << kullanilanIndex << endl << endl;

    float intVeFloatFloatDegeri = get<1>(intVeFloat);
    cout << "intVeFloat Variant'nin 1. Index'deki turunun degeri; " << intVeFloatFloatDegeri << endl << endl;

    return 0;
}
