Veritabanı İşlemleri

Qt'de Veritabanı

Qt, devasa bir ekosistem olduğu için tabikide veritabanı konusunda da pakete sahiptir. Ben Standart C++ programlama dilinde kullandığım veritabanı sistemi olan SQLite'ı kullanıcağım. Konumuz veritabanı işlemleri olduğu için "veritabaniIslemleri" adlı bir form projemizi açalım.
Paketi eklemek için;
            
CMakeLists.txt

cmake_minimum_required(VERSION 3.16)

project(veritabaniIslemleri VERSION 0.1 LANGUAGES CXX)

set(CMAKE_AUTOUIC ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

find_package(QT NAMES Qt6 Qt5 REQUIRED COMPONENTS Widgets)
find_package(Qt${QT_VERSION_MAJOR} REQUIRED COMPONENTS Widgets)
find_package(Qt6 REQUIRED COMPONENTS Sql) # <-

set(PROJECT_SOURCES
        main.cpp
        mainwindow.cpp
        mainwindow.h
        mainwindow.ui
)

if(${QT_VERSION_MAJOR} GREATER_EQUAL 6)
    qt_add_executable(veritabaniIslemleri
        MANUAL_FINALIZATION
        ${PROJECT_SOURCES}
    )
# Define target properties for Android with Qt 6 as:
#    set_property(TARGET veritabaniIslemleri APPEND PROPERTY QT_ANDROID_PACKAGE_SOURCE_DIR
#                 ${CMAKE_CURRENT_SOURCE_DIR}/android)
# For more information, see https://doc.qt.io/qt-6/qt-add-executable.html#target-creation
else()
    if(ANDROID)
        add_library(veritabaniIslemleri SHARED
            ${PROJECT_SOURCES}
        )
# Define properties for Android with Qt 5 after find_package() calls as:
#    set(ANDROID_PACKAGE_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/android")
    else()
        add_executable(veritabaniIslemleri
            ${PROJECT_SOURCES}
        )
    endif()
endif()

target_link_libraries(veritabaniIslemleri PRIVATE Qt${QT_VERSION_MAJOR}::Widgets)
target_link_libraries(veritabaniIslemleri PRIVATE Qt6::Sql) # <-

# Qt for iOS sets MACOSX_BUNDLE_GUI_IDENTIFIER automatically since Qt 6.1.
# If you are developing for iOS or macOS you should consider setting an
# explicit, fixed bundle identifier manually though.
if(${QT_VERSION} VERSION_LESS 6.1.0)
  set(BUNDLE_ID_OPTION MACOSX_BUNDLE_GUI_IDENTIFIER com.example.veritabaniIslemleri)
endif()
set_target_properties(veritabaniIslemleri PROPERTIES
    ${BUNDLE_ID_OPTION}
    MACOSX_BUNDLE_BUNDLE_VERSION ${PROJECT_VERSION}
    MACOSX_BUNDLE_SHORT_VERSION_STRING ${PROJECT_VERSION_MAJOR}.${PROJECT_VERSION_MINOR}
    MACOSX_BUNDLE TRUE
    WIN32_EXECUTABLE TRUE
)

include(GNUInstallDirs)
install(TARGETS veritabaniIslemleri
    BUNDLE DESTINATION .
    LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR}
    RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR}
)

if(QT_VERSION_MAJOR EQUAL 6)
    qt_finalize_executable(veritabaniIslemleri)
endif()

Deskteklenin Veritabanları

Qt'nin Sql paketinde birçok veritabanını desteklemektedir. Bunları aşağıdaki listede bulabilirsiniz.
  • DB2 (IBM DB2)
  • InterBase
  • MYSQL / MariaDB
  • Oracle
  • ODBC
  • PostgreSQL
  • SQLite
  • Mimer SQL

Veritabanı Oluşturma

Artık veritabanı oluşturabiliriz... Aşağıdaki komutlar ile veritabanımızı oluştralım;
mainwindow.h

#ifndef MAINWINDOW_H
#define MAINWINDOW_H

#include <QMainWindow>
#include <QMessageBox>
#include <QSql>         //
#include <QSqlDatabase> // Kütüphanemiz
#include <QSqlError>    //

QT_BEGIN_NAMESPACE
namespace Ui {
class MainWindow;
}
QT_END_NAMESPACE

class MainWindow : public QMainWindow
{
    Q_OBJECT

public:
    explicit MainWindow(QWidget *parent = nullptr);
    ~MainWindow() override;

public slots:
    void pushButton_clicked(); // Veritabanı oluşturma

private:
    Ui::MainWindow *ui;
};
#endif // MAINWINDOW_H


mainwindow.cpp

#include "mainwindow.h"
#include "./ui_mainwindow.h"

QSqlDatabase sqliteVeritabani; // Global olarak tanımladık

MainWindow::MainWindow(QWidget *parent)
    : QMainWindow(parent)
    , ui(new Ui::MainWindow)
{
    ui->setupUi(this);

    connect(ui->pushButton, &QPushButton::clicked, this, &MainWindow::pushButton_clicked);
}

MainWindow::~MainWindow()
{
    delete ui;
}

void MainWindow::pushButton_clicked()
{
    sqliteVeritabani = QSqlDatabase::addDatabase("QSQLITE"); // Veritabanımızın ne tür olucağını belirledik
    // "QSQLITE"; Bu ifade yerine yukarıda verdiğimiz veritabanlarını isimlerini Qt'deki karşılıklarına denk gelen ifadeyi yazınız. Ben Sqlite kullanıcağım için "QSQLITE" olarak yazdım
    sqliteVeritabani.setDatabaseName("veritabani.db"); // Bu kısımda ise veritabanımızın ismini belirliyoruz. Aslında bu kısım SQL dilinde "create database veritabani" komutuna karşılık gelir

    if(sqliteVeritabani.open()) // Eğer tabanı açılırsa
    {
        QMessageBox(QMessageBox::Icon::Information, "BAŞARILI", "Veritabanı başarılı bir şekilde açıldı!").exec();
    }
    else // açılamazsa
    {
        QMessageBox(QMessageBox::Icon::Critical, "BAŞARISIZ!", QString("Sqlite veritabanı açılamadı: ").append(sqliteVeritabani.lastError().text())).exec();
        // Eğer bir durumda hata alınır ise
        //  sqliteVeritabani.lastError().text()
        // komutu ile hatanın mesajı alınabilir
    }
}

Tablo Oluşturma ve Silme

Tablomuz basit olucak. Veri olarak id, isim ve soyisim tutucak. Aşağıdaki örnekte inceleyelim;
mainwindow.h

#ifndef MAINWINDOW_H
#define MAINWINDOW_H

#include <QMainWindow>
#include <QMessageBox>
#include <QSql>         //
#include <QSqlDatabase> // Kütüphanemiz
#include <QSqlError>    //
#include <QSqlQuery>    // Sorgu için kütüphanemiz

QT_BEGIN_NAMESPACE
namespace Ui {
class MainWindow;
}
QT_END_NAMESPACE

class MainWindow : public QMainWindow
{
    Q_OBJECT

public:
    explicit MainWindow(QWidget *parent = nullptr);
    ~MainWindow() override;

public slots:
    void pushButton_clicked(); // Veritabanı oluşturma
    void pushButton2_clicked(); // Tablo oluşturma
    void pushButton3_clicked(); // Tabloyu silme

private:
    Ui::MainWindow *ui;
};
#endif // MAINWINDOW_H


mainwindow.cpp

#include "mainwindow.h"
#include "./ui_mainwindow.h"

QSqlDatabase sqliteVeritabani; // Global olarak tanımladık

MainWindow::MainWindow(QWidget *parent)

    connect(ui->pushButton, &QPushButton::clicked, this, &MainWindow::pushButton_clicked);

void MainWindow::pushButton2_clicked()
{
    QSqlQuery sorgu1(sqliteVeritabani); // Sql sorgu nesnemiz. Parantezlerine ise sorgudan etkilencek olan veritabanı nesnesini yazdık

    if(sorgu1.exec()) // Sorgumuzu çalıştırdık
    { // Başarılı ise
        QMessageBox(QMessageBox::Icon::Information, "BAŞARILI", "Tablomuz başarılı bir şekilde oluşturuldu!").exec();
    }
    else // değil ise

        //  sorgu1.lastError().text()
        // komutu ile alınır

}

    {
        QMessageBox(QMessageBox::Icon::Critical, "BAŞARISIZ", QString("Tablomuz silinemedi: ").append(sorgu1.lastError().text())).exec();
    }
}

void MainWindow::pushButton4_clicked()
{
    QSqlQuery sorgu1(sqliteVeritabani);
    sorgu1.prepare("insert into kisiler(id, ad, soyad) values((:id), (:ad), (:soyad))"); // Güvenli sorgu yazdık
    sorgu1.bindValue(":id", ui->lineEdit->text().toInt()); // Güvenli sorgu argümanları doldurduk
    // ":id"; Argüman ismi
    // ui->lineEdit->text().toInt(); Argümana yazılcak olan değer
    sorgu1.bindValue(":ad", ui->lineEdit_2->text());
    sorgu1.bindValue(":soyad", ui->lineEdit_3->text());

    if(sorgu1.exec())
    {
        QMessageBox(QMessageBox::Icon::Information, "BAŞARILI", "Veri başarlı bir şekilde eklendi!").exec();
    }
    else
    {
        QMessageBox(QMessageBox::Icon::Critical, "BAŞARISIZ", QString("Veri eklenemedi: ").append(sorgu1.lastError().text())).exec();
    }
}

void MainWindow::pushButton5_clicked()
{
    QSqlQuery sorgu1(sqliteVeritabani);
    sorgu1.prepare("delete from kisiler where id=(:id)");
    sorgu1.bindValue(":id", ui->lineEdit_4->text().toInt());

    if(sorgu1.exec())
    {
        QMessageBox(QMessageBox::Icon::Information, "BAŞARILI", "Veri başarlı bir şekilde silinidi!").exec();
    }
    else
    {
        QMessageBox(QMessageBox::Icon::Critical, "BAŞARISIZ", QString("Veri silinemedi: ").append(sorgu1.lastError().text())).exec();
    }
}

void MainWindow::pushButton6_clicked()
{
    QSqlQuery sorgu1(sqliteVeritabani);
    sorgu1.prepare("select * from kisiler where id=(:id)");
    sorgu1.bindValue(":id", ui->lineEdit_5->text().toInt());

    if(sorgu1.exec()) // Başarılı ise
    {
        while(sorgu1.next()) // Select komutu bizlere geri dönüşlü bir değer döndürdüğünden, while döngüsü ile verileri alırız
        {
            ui->lineEdit_7->setText(sorgu1.value(0).toString());
            // Sorgudan bir çıktı geliyor ise
            //  sorgu1.value(int index)
            // komutu ile alınabilir
            ui->lineEdit_6->setText(sorgu1.value(1).toString());
            ui->lineEdit_8->setText(sorgu1.value(2).toString());
        }
    }
    else
    {
        QMessageBox(QMessageBox::Icon::Critical, "BAŞARISIZ", QString("Kullanıcı bulunamadı: ").append(sorgu1.lastError().text())).exec();
    }
}

void MainWindow::pushButton7_clicked()
{
    QSqlQuery sorgu1(sqliteVeritabani);
    sorgu1.prepare("update kisiler set ad=(:ad), soyad=(:soyad) where id=(:id)");
    sorgu1.bindValue(":id", ui->lineEdit_7->text().toInt());
    sorgu1.bindValue(":ad", ui->lineEdit_6->text());
    sorgu1.bindValue(":soyad", ui->lineEdit_8->text());

    if(sorgu1.exec())
    {
        QMessageBox(QMessageBox::Icon::Information, "BAŞARILI", "Veri başarlı bir şekilde güncellendi!").exec();
    }
    else
    {
        QMessageBox(QMessageBox::Icon::Critical, "BAŞARISIZ", QString("Veri güncellenemedi: ").append(sorgu1.lastError().text())).exec();
    }
}

void MainWindow::pushButton8_clicked()
{
    QSqlQuery sorgu1(sqliteVeritabani);
    sorgu1.prepare("select * from kisiler");

    if(sorgu1.exec())
    {
        QString sorguCiktisi = "";

        while(sorgu1.next())
        {
            sorguCiktisi.append("id: ").append(sorgu1.value(0).toString()).append(" | ")
                        .append("ad: ").append(sorgu1.value(1).toString()).append(" | ")
                        .append("soyad: ").append(sorgu1.value(2).toString())
                        .append("\n");
        }

        ui->textEdit->setText(sorguCiktisi);
    }
    else
    {
        QMessageBox(QMessageBox::Icon::Critical, "BAŞARISIZ", QString("Kullanıcı bulunamadı: ").append(sorgu1.lastError().text())).exec();
    }
}
