Ana Pencere

QMainWindow Nedir?

QMainWindow, uygulamanın ilk olarak açılan penceresidir. Tüm bileşenler ana pencere yani QMainWindow sınıfında yer alır. Bu sayfada QMainWindow'dun yani ana pencerenin özelliklerini göreceğiz.
Bu sayfada ana pencere ilgili özellikleri görüceğimiz için "anaPencere" adlı bir proje oluşturalım.

QMenuBar

QMenuBar, pencerenin üst kısmında liste vari bir seçenek havuzu sunan yapıdır. İçerisine QMenu ve QMenu'lerin içlerinde QAction'ların kullanıcı ile etkileşime girebilmesini sağlar. Benzetme açısından çoğu uygulamada yukarıda çıkan "File" veya "Project" gibi kısımlar QMenuBar'dır.
Peki QMenuBar niçin kullanılır? Genellikle Meta işlemler için kategörlü bir işlemler bütünü sağlar. Örnek olarak "File" yani dosya kısmında "Open File" işlemi ile bilgisayardan dosya açmamızı sağlar. Bu örnekte  QMenuBar "File", QAction ise "Open File" olur.
mainwindow.h

#ifndef MAINWINDOW_H
#define MAINWINDOW_H

#include <QMainWindow>

QT_BEGIN_NAMESPACE
namespace Ui {
class MainWindow;
}
QT_END_NAMESPACE

class MainWindow : public QMainWindow
{
    Q_OBJECT

public:
    explicit MainWindow(QWidget *parent = nullptr);
    ~MainWindow() override;

public slots:
    void dosyaAcmaSecenegi(bool); // QAction -> triggered
    void dosyaOkumaSecenegi(bool); // QAction -> triggered

private:
    Ui::MainWindow *ui;
};
#endif // MAINWINDOW_H


mainwindow.cpp

#include "mainwindow.h"
#include "./ui_mainwindow.h"

MainWindow::MainWindow(QWidget *parent)
    : QMainWindow(parent)
    , ui(new Ui::MainWindow)
{
    ui->setupUi(this);

    connect(ui->actionDosya_A_3, &QAction::triggered, this, &MainWindow::dosyaAcmaSecenegi);
    connect(ui->actionDosya_Oku, &QAction::triggered, this, &MainWindow::dosyaOkumaSecenegi);
}

MainWindow::~MainWindow()
{
    delete ui;
}

void MainWindow::dosyaAcmaSecenegi(bool deger)
{
    qDebug() << "Kullanıcı dosya açmak istiyor" << Qt::endl;
}

void MainWindow::dosyaOkumaSecenegi(bool deger)
{
    qDebug() << "Kullanıcı dosya okumak istiyor" << Qt::endl;
}

QToolBar

QToolBar, içerisine ögeler ekliyebilceğimiz pencerecik olarak görev yapar. QToolBar'ı başındaki simgeden tutup ana pencerenin istediğim yerine koyabilceğimiz, kendisine özel bir Layout düzeni olan bir pencerecik elemanıdır. İçerisine öge ekleme işlemi kod tarafında yapılır.
mainwindow.cpp

#include "mainwindow.h"
#include "./ui_mainwindow.h"

#include <QPushButton>
#include <QLabel>
#include <QLineEdit>

MainWindow::MainWindow(QWidget *parent)
    : QMainWindow(parent)
    , ui(new Ui::MainWindow)
{
    ui->setupUi(this);

    QPushButton *button1 = new QPushButton(this); // QPushButton oluşturma
    button1->setText("Bana bas");
    ui->toolBar->addWidget(button1); // Toolbar'a eleman ekleme
    QLabel *label1 = new QLabel("Bu bir QLabel"); // QLabel oluşturma
    ui->toolBar->addWidget(label1);
    QLineEdit *edit1 = new QLineEdit(this); // QLineEdit oluşturma
    ui->toolBar->addWidget(edit1);

    connect(ui->actionDosya_A_3, &QAction::triggered, this, &MainWindow::dosyaAcmaSecenegi);
    connect(ui->actionDosya_Oku, &QAction::triggered, this, &MainWindow::dosyaOkumaSecenegi);
}

MainWindow::~MainWindow()
{
    delete ui;
}

void MainWindow::dosyaAcmaSecenegi(bool deger)
{
    qDebug() << "Kullanıcı dosya açmak istiyor" << Qt::endl;
}

void MainWindow::dosyaOkumaSecenegi(bool deger)
{
    qDebug() << "Kullanıcı dosya okumak istiyor" << Qt::endl;
}

QStatusBar

QStatusBar, ana pencerenin alt kısmında yer alan ve pencere hakkında mesaj yazdırabilceğimiz bir ögedir. Mesaj yazma işlemi kod tarafında gerçekleştirilir.
mainwindow.cpp

#include "mainwindow.h"
#include "./ui_mainwindow.h"

#include <QPushButton>
#include <QLabel>
#include <QLineEdit>

MainWindow::MainWindow(QWidget *parent)
    : QMainWindow(parent)
    , ui(new Ui::MainWindow)
{
    ui->setupUi(this);

    QPushButton *button1 = new QPushButton(this); // QPushButton oluşturma
    button1->setText("Bana bas");
    ui->toolBar->addWidget(button1); // Toolbar'a eleman ekleme
    QLabel *label1 = new QLabel("Bu bir QLabel"); // QLabel oluşturma
    ui->toolBar->addWidget(label1);
    QLineEdit *edit1 = new QLineEdit(this); // QLineEdit oluşturma
    ui->toolBar->addWidget(edit1);

    connect(ui->actionDosya_A_3, &QAction::triggered, this, &MainWindow::dosyaAcmaSecenegi);
    connect(ui->actionDosya_Oku, &QAction::triggered, this, &MainWindow::dosyaOkumaSecenegi);

    connect(ui->pushButton_2, &QPushButton::clicked, this,
    [this]()
    {
        ui->statusbar->showMessage("Selamlar Herkese!");
    });
}

QDockWidget

QDockWidget, QToolBar'ın daha da gelişmiş bir versiyonudur. Kullanıcı tarafından küçültebilir veya büyültebilir, ana pencerenin dışında pencere olarak kullanılabilir ve istenilen Layout atanabilir bir ögedir.

Merkezi Widget Ayarlama

Merkezi Widget, ana pencerenin görüntülediği ana Widget'tır.
mainwindow.cpp

    : QMainWindow(parent)
    , ui(new Ui::MainWindow)
{
    ui->setupUi(this);

    QPushButton *button1 = new QPushButton(this); // QPushButton oluşturma
    button1->setText("Bana bas");
    ui->toolBar->addWidget(button1); // Toolbar'a eleman ekleme
    QLabel *label1 = new QLabel("Bu bir QLabel"); // QLabel oluşturma
    ui->toolBar->addWidget(label1);
    QLineEdit *edit1 = new QLineEdit(this); // QLineEdit oluşturma
    ui->toolBar->addWidget(edit1);

    connect(ui->actionDosya_A_3, &QAction::triggered, this, &MainWindow::dosyaAcmaSecenegi);

Kısayol Atama

QAction ile ana penceremize klavyeden kısayollar atayabiliriz. Kısayol atama işlemi kod tarafında gerçekleştirilir.
mainwindow.cpp

#include "mainwindow.h"
#include "./ui_mainwindow.h"

#include <QPushButton>
#include <QLabel>
#include <QLineEdit>
#include <QShortcut> // Kısayol kütüphanemiz

MainWindow::MainWindow(QWidget *parent)
    : QMainWindow(parent)
    , ui(new Ui::MainWindow)
{
    ui->setupUi(this);

    QPushButton *button1 = new QPushButton(this); // QPushButton oluşturma
    button1->setText("Bana bas");
    ui->toolBar->addWidget(button1); // Toolbar'a eleman ekleme
    QLabel *label1 = new QLabel("Bu bir QLabel"); // QLabel oluşturma
    ui->toolBar->addWidget(label1);
    QLineEdit *edit1 = new QLineEdit(this); // QLineEdit oluşturma
    ui->toolBar->addWidget(edit1);

    connect(ui->actionDosya_A_3, &QAction::triggered, this, &MainWindow::dosyaAcmaSecenegi);
    connect(ui->actionDosya_Oku, &QAction::triggered, this, &MainWindow::dosyaOkumaSecenegi);

    connect(ui->pushButton_2, &QPushButton::clicked, this,
    [this]()
    {
        ui->statusbar->showMessage("Yeni bir merkezi Widget seçildi!");

        this->setCentralWidget(ui->widget); // Merkezi Widget ayarlama
    });

    QShortcut *kisayol_YukariOk = new QShortcut(QKeySequence(Qt::Key_Up), this); // Kısayol oluşturuyoruz
    // Qt::Key_Up; Klavyedeki tuşun Qt isim alanındaki karşılığı. Bu durumda yukarı ok oluyor
    connect(kisayol_YukariOk, &QShortcut::activated, this,
    [this]()
    {
        this->setCentralWidget(ui->widget);
    }); // Bağlıyoruz
}

{
    delete ui;
}

{
    qDebug() << "Kullanıcı dosya açmak istiyor" << Qt::endl;

Ana Pencerenin Özellikleri

Ana pencerenin bir çok özelliği istersek kod tarafında değiştirebiliriz. İkonu, başlığı veya boyutu gibi özellikleri rahatlıkla değiştirelebilir.

#include "./ui_mainwindow.h"

#include <QLabel>
#include <QLineEdit>
#include <QShortcut>

MainWindow::MainWindow(QWidget *parent)
    : QMainWindow(parent)
    , ui(new Ui::MainWindow)
{

    ui->toolBar->addWidget(label1);

    ui->toolBar->addWidget(edit1);

    connect(ui->actionDosya_A_3, &QAction::triggered, this, &MainWindow::dosyaAcmaSecenegi);
    connect(ui->actionDosya_Oku, &QAction::triggered, this, &MainWindow::dosyaOkumaSecenegi);

    connect(ui->pushButton_2, &QPushButton::clicked, this,
    [this]()
    {
        ui->statusbar->showMessage("Yeni bir merkezi Widget seçildi!");

        this->setCentralWidget(ui->widget); // Merkezi Widget ayarlama

    QShortcut *kisayol_YukariOk = new QShortcut(QKeySequence(Qt::Key_Up), this); // Kısayol oluşturuyoruz

    [this]()
    {
        this->setCentralWidget(ui->widget);
    }); // Bağlıyoruz

    // this işaretçisi bizim ana penceremiz olan MainWindow'u temsil eder

    this->setWindowIcon(QIcon("C:\\Users\\DATA\\Pictures\\workAndData.png")); // Pencerenin ikonunu ayarlama

    this->setWindowTitle("Ana Pencere Özellikleri"); // Ana pencerenin başlığını ayarlar
