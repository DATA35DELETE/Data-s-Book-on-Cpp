Test Sistemi

Modül Dahil Etme

Qt'nin Test modülünü inceleyelim. Bunun için "test" adında bir Widget projesi oluşturalım. Modülü dahil etmek için;
CMakeLists.txt

find_package(Qt6 REQUIRED COMPONENTS Test)
target_link_libraries(test PRIVATE Qt::Test)

QMake

QT += testlib

QTest

QTest, Qt'nin test modülündeki ana sınıftır. Test işlemleri bu sınıf üzerinde gerçekleştirilir. Bu sınıf içerisinde QVERIFY ve QCOMPARE makrolarını sunar. QVERIFY temel bir test çıktısı verirken QCOMPARE daha detaylı bir şekilde test çıktısı vermektedir.
Örnek olması açısından uygulama başladığında doğru ve yanlış, QVERIFY ve QCOMPARE testlerini çalıştıran bir uygulama yazalım.
Bunun için hangi akla hizmet bilinmez fakat testler ayrı bir sınıf dosyasına yazılmak zorundadır. Bu durum sebebiyle "testler" adlı bir sınıf oluşturalım.

testler.h

#ifndef TESTLER_H
#define TESTLER_H

#include <QObject>
#include <QTest>

class testler
{
    Q_OBJECT

private:
    int a = 100; // Herhangi bir değişken

private slots:
    void qVerifyDogruTest();
    void qVerifyYanlisTest();

    void qCompareDogruTest();
    void CompareYanlisTest();
};

#endif // TESTLER_H


testler.cpp

#include "testler.h"

void testler::qVerifyDogruTest()
{
    QVERIFY(a / 10 == 10); // Bu kısımda test mantığı şu şekilde çalışır. QVERIFY aslında ifadeye bakar. Eğer ifadenin sonunu 1 ise test doğru olur. Fakat ifade 0 ise içerisine yazılan ifadeyle bir test çıktısı gönderir
}
void testler::qVerifyYanlisTest()
{
    QVERIFY(a / 5 == 40);
}

void testler::qCompareDogruTest()
{
    QCOMPARE(a / 10, 10); // QCOMPARE farklı olarak bizden 2 parametreli bir yapı ister. İlk parametre ifade, diyeri ise olması gerekn sonucu temsil eder
}
void testler::CompareYanlisTest()
{
    QCOMPARE(a / 5, 40);
}

mainwindow.h

#ifndef MAINWINDOW_H
#define MAINWINDOW_H

#include "testler.h" // Sınıfın başlık dosyasını dahil ettik

#include <QMainWindow>
#include <QTest> // Kütüphanemiz

QT_BEGIN_NAMESPACE
namespace Ui {
class MainWindow;
}
QT_END_NAMESPACE

class MainWindow : public QMainWindow
{
    Q_OBJECT

public:
    explicit MainWindow(QWidget *parent = nullptr);
    ~MainWindow() override;

private:
    Ui::MainWindow *ui;
    testler test; // Nesnemizi oluşturduk
};
#endif // MAINWINDOW_H


mainwindow.cpp

#include "mainwindow.h"
#include "ui_mainwindow.h"

MainWindow::MainWindow(QWidget *parent)
    : QMainWindow(parent)
    , ui(new Ui::MainWindow)
{
    ui->setupUi(this);

    QTest::qExec(&test); // Testi çalıştırdık
}

MainWindow::~MainWindow()
{
    delete ui;
}

QBENCHMARK

QBENCHMARK, işlemin hızını ölçer.

testler.cpp

#include "testler.h"

void testler::qVerifyDogruTest()
{
    QVERIFY(a / 10 == 10);
}
void testler::qVerifyYanlisTest()
{
    QVERIFY(a / 5 == 40);
}

void testler::qCompareDogruTest()
{
    QCOMPARE(a / 10, 10);
}
void testler::CompareYanlisTest()
{
    QCOMPARE(a / 5, 40);
}

void testler::qBenchmarkTest()
{
    QBENCHMARK(a / 10); // QVERIFY gibi yazılır
}

Tıklama Testi

QTest sınıfı, bizlere fare tıklamasını simüle eden ve test etirebilmemizi sağlayan bir işlev sunar. QTest::mouseClick ile bu test gerçekleştirilebilir.
mainwindow.cpp

#include "mainwindow.h"
#include "ui_mainwindow.h"

MainWindow::MainWindow(QWidget *parent)
    : QMainWindow(parent)
    , ui(new Ui::MainWindow)
{
    ui->setupUi(this);

    QTest::qExec(&test);

    connect(ui->pushButton, &QPushButton::clicked, this, [](){qDebug() << "Butona tıklandı!";}); // Test edilcek butona bir işlev ekledik

    QTest::mouseClick(ui->pushButton, Qt::MouseButton::LeftButton, Qt::NoModifier, QPoint(10,10), 100); // Tıklama testi
    // ui->pushButton; Test edilcek Widget
    // Qt::MouseButton::LeftButton; Hangi fare butonu
    // Qt::NoModifier; Düzenleme
    // QPoint(10,10); Widget üzerinde hangi noktada tıklanıcak
    // 100; Tıklama için gecikme
}

MainWindow::~MainWindow()
{
    delete ui;
}

QSignalSpy

QSignalSpy; herhangi bir Widget'ın herhangi bir sinyaline bağlanarak, o sinyali izliyip analiz eden bir sınıftır.
mainwindow.h

#ifndef MAINWINDOW_H
#define MAINWINDOW_H

#include "testler.h"

#include <QMainWindow>
#include <QMessageBox>
#include <QTest>
#include <QSignalSpy> // Kütüphanemiz

QT_BEGIN_NAMESPACE
namespace Ui {
class MainWindow;
}
QT_END_NAMESPACE

class MainWindow : public QMainWindow
{
    Q_OBJECT

public:
    explicit MainWindow(QWidget *parent = nullptr);
    ~MainWindow() override;

private:
    Ui::MainWindow *ui;
    testler test;
};
#endif // MAINWINDOW_H


mainwindow.cpp

#include "mainwindow.h"
#include "ui_mainwindow.h"

QSignalSpy *spy; // Sinyal izlicimiz

MainWindow::MainWindow(QWidget *parent)
    : QMainWindow(parent)
    , ui(new Ui::MainWindow)
{
    ui->setupUi(this);

    //QTest::qExec(&test);

    //QTest::mouseClick(ui->pushButton, Qt::MouseButton::LeftButton, Qt::NoModifier, QPoint(10,10), 100);

    spy = new QSignalSpy(ui->pushButton, SIGNAL(clicked(bool))); // Nesnesini oluşturduk
    // ui->pushButton; İzlenicek Widget
    // SIGNAL(clicked(bool)); İzlenen Widget'ın sinyali

    connect(ui->pushButton, &QPushButton::clicked, this, []()
    {
        qDebug() << spy->count(); // Miktarını alıp yazdırıyoruz
    });
}

MainWindow::~MainWindow()
{
    delete ui;
}
